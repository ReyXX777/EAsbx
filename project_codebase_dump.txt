# agent_engine.py
import asyncio
import os
from PySide6.QtCore import QThread, Signal
from browser_use import Agent, Browser, ChatGoogle

from agent_state_manager import AgentStateManager


class AgentExecutionWorker(QThread):
    step_updated = Signal(str)
    task_finished = Signal(str)
    error_occurred = Signal(str)

    def __init__(
        self,
        task_description,
        api_key,
        headless=False,
        cdp_url="http://127.0.0.1:9222",
    ):
        super().__init__()
        self.task_description = task_description
        self.api_key = api_key
        self.headless = headless
        self.cdp_url = cdp_url

        # Delegate state tracking to independent state manager
        self.state_manager = AgentStateManager()

    def get_status(self) -> dict:
        """API/UI Endpoint method to query current agent state at any time."""
        return self.state_manager.get_status()

    async def _on_step_end_callback(self, agent: Agent):
        """Lifecycle hook executed after every single agent step."""
        formatted_step_msg = self.state_manager.record_step(agent)
        self.step_updated.emit(formatted_step_msg)

    def run(self):
        # Create a clean asyncio event loop for this QThread
        loop = asyncio.new_event_loop()
        asyncio.set_event_loop(loop)
        try:
            loop.run_until_complete(self._execute_agent_task())
        except Exception as e:
            err_msg = str(e)
            self.state_manager.mark_failed(err_msg)
            self.error_occurred.emit(err_msg)
        finally:
            loop.close()

    async def _execute_agent_task(self):
        try:
            self.state_manager.start_task()

            # Pass API key into environment for ChatGoogle
            os.environ["GOOGLE_API_KEY"] = self.api_key

            # Initialize LLM with high-throughput free-tier model
            llm = ChatGoogle(model="gemini-3.5-flash-lite")

            # Connect to existing embedded Qt WebEngine browser via CDP if provided
            if self.cdp_url:
                browser = Browser(cdp_url=self.cdp_url)
            else:
                browser = Browser(headless=self.headless)

            # Instantiate Agent with fail-safes and step hooks
            agent = Agent(
                task=self.task_description,
                llm=llm,
                browser=browser,
                llm_timeout=30,  # Max 30s for LLM API response
                step_timeout=60,  # Max 60s per DOM action
                max_failures=3,  # Max consecutive retries on error
                final_response_after_failure=True,  # Return best-effort summary if max_failures reached
            )

            self.step_updated.emit(
                f"🚀 Initializing Agent for task: '{self.task_description}'"
            )

            # Run loop with step monitoring callback
            result = await agent.run(on_step_end=self._on_step_end_callback)

            self.state_manager.mark_completed()
            self.task_finished.emit(str(result))

        except asyncio.TimeoutError:
            err_msg = "Agent execution timed out while waiting for model or browser response."
            self.state_manager.mark_failed(err_msg)
            self.error_occurred.emit(err_msg)

        except Exception as e:
            err_msg = f"Agent Execution Failure: {str(e)}"
            self.state_manager.mark_failed(err_msg)
            self.error_occurred.emit(err_msg)

# agent_response_parser.py
import html
import re
from typing import Dict, List, Tuple


class AgentResponseParser:
    """Predictive parsing engine for browser-use agent outputs."""

    @classmethod
    def parse_and_format(cls, raw_result_str: str) -> str:
        """Parses raw AgentHistoryList string or plain text into formatted HTML

        with cascading styles and collapsible step logs.
        """
        final_summary, extracted_steps = cls._extract_semantics(raw_result_str)

        # 1. Format the main output card
        formatted_summary = cls._format_summary_card(final_summary)

        # 2. Format step telemetry into a collapsible HTML block
        formatted_steps = cls._format_collapsible_steps(extracted_steps)

        return f"{formatted_summary}\n{formatted_steps}"

    @classmethod
    def _extract_semantics(
        cls, raw_str: str
    ) -> Tuple[str, List[Dict[str, str]]]:
        """Predictively extracts final outcome and intermediate execution steps."""
        steps = []
        final_summary = ""

        # Regex patterns to capture intermediate ActionResults
        action_pattern = re.compile(
            r"ActionResult\([^)]*?extracted_content=['\"](.*?)['\"]", re.DOTALL
        )
        for match in action_pattern.finditer(raw_str):
            try:
                content = match.group(1).encode("utf-8").decode("unicode_escape")
            except Exception:
                content = match.group(1)

            if content.strip() and not content.startswith("Match:"):
                clean_content = (
                    content[:120] + "..." if len(content) > 120 else content
                )
                steps.append(
                    {"type": "Action / DOM Step", "details": clean_content}
                )

        # Pattern for final text inside done(...)
        done_pattern = re.compile(
            r"done=['\"]?\{'text':\s*['\"](.*?)['\"]", re.DOTALL
        )
        done_match = done_pattern.search(raw_str)

        if done_match:
            try:
                final_summary = (
                    done_match.group(1)
                    .encode("utf-8")
                    .decode("unicode_escape")
                )
            except Exception:
                final_summary = done_match.group(1)
        else:
            # Fallback for extracted content near end of response
            summary_pattern = re.compile(
                r"extracted_content=['\"](Match:[^'\"]+)['\"]", re.DOTALL
            )
            summary_match = summary_pattern.search(raw_str)
            if summary_match:
                final_summary = summary_match.group(1)
            else:
                final_summary = raw_str

        return final_summary, steps

    @classmethod
    def _format_summary_card(cls, text: str) -> str:
        """Renders the main agent output inside a visual CSS container."""
        if not text:
            text = "Task executed successfully with no returned text."

        # Escape HTML characters first to prevent XSS/broken layout
        text_escaped = html.escape(text)

        # Convert simple markdown to HTML inside summary card
        # Bold
        text_escaped = re.sub(
            r"\*\*(.*?)\*\*", r"<strong>\1</strong>", text_escaped
        )

        # Bullets (- Item)
        lines = text_escaped.split("\n")
        formatted_lines = []
        for line in lines:
            line_str = line.strip()
            if line_str.startswith("- ") or line_str.startswith("* "):
                formatted_lines.append(
                    f"<li style='margin-left: 15px;'>{line_str[2:]}</li>"
                )
            elif line_str:
                formatted_lines.append(f"<div>{line_str}</div>")

        processed_body = "".join(formatted_lines)

        return f"""
        <div style="background-color: #f8fafc; border-left: 4px solid #16a34a; padding: 10px 12px; margin-top: 6px; margin-bottom: 8px; border-radius: 4px;">
            <div style="font-weight: bold; color: #1e293b; font-size: 13px; margin-bottom: 6px;">📌 Execution Result</div>
            <div style="font-size: 13px; color: #334155; line-height: 1.5;">{processed_body}</div>
        </div>
        """

    @classmethod
    def _format_collapsible_steps(cls, steps: List[Dict[str, str]]) -> str:
        """Generates an HTML details/summary collapsible dropdown for agent steps."""
        if not steps:
            return ""

        step_items_html = ""
        for idx, step in enumerate(steps, 1):
            clean_details = html.escape(step["details"])
            step_items_html += f"""
            <li style="margin-bottom: 4px;">
                <strong style="color: #64748b;">Step {idx}:</strong> <code>{clean_details}</code>
            </li>
            """

        return f"""
        <details style="margin-top: 6px; background-color: #f1f5f9; border: 1px solid #cbd5e1; border-radius: 4px; padding: 6px 10px; font-size: 12px;">
            <summary style="font-weight: bold; color: #475569; cursor: pointer; user-select: none;">
                🔍 View Agent Execution Steps ({len(steps)} actions)
            </summary>
            <ul style="margin: 8px 0 4px 18px; padding: 0; color: #334155; font-family: monospace;">
                {step_items_html}
            </ul>
        </details>
        """

# agent_state_manager.py
from typing import Any, Dict, Optional


class AgentStateManager:
    """Standalone, testable state monitor for browser agent execution."""

    def __init__(self):
        self._state: Dict[str, Any] = {
            "status": "idle",  # "idle", "running", "completed", "failed"
            "current_step": 0,
            "last_action": "Not started",
            "error": None,
        }

    def start_task(self) -> None:
        """Resets and transitions state to running."""
        self._state["status"] = "running"
        self._state["current_step"] = 0
        self._state["last_action"] = "Initializing..."
        self._state["error"] = None

    def record_step(self, agent_instance: Optional[Any] = None) -> str:
        """Parses the latest action from the agent history and increments step count."""
        self._state["current_step"] += 1
        last_action_desc = "Processing step..."

        if agent_instance and hasattr(agent_instance, "history"):
            history = agent_instance.history
            if history and hasattr(history, "model_actions") and history.model_actions():
                last_action_desc = str(history.model_actions()[-1])

        self._state["last_action"] = last_action_desc
        return f"Step {self._state['current_step']}: {last_action_desc[:80]}..."

    def mark_completed(self) -> None:
        """Transitions state to completed."""
        self._state["status"] = "completed"

    def mark_failed(self, error_message: str) -> None:
        """Transitions state to failed with error details."""
        self._state["status"] = "failed"
        self._state["error"] = error_message

    def get_status(self) -> Dict[str, Any]:
        """Returns a copy of the current state snapshot for UI / API endpoints."""
        return self._state.copy()

from PySide6.QtWidgets import QMenuBar, QMessageBox, QWidget


class AppMenuBar(QMenuBar):
    """Custom Application Menu Bar providing agent diagnostics and status monitoring."""

    def __init__(self, parent: QWidget = None):
        super().__init__(parent)
        self.main_window = parent
        self._init_menus()

    def _init_menus(self):
        # 1. Agent / Status Menu
        agent_menu = self.addMenu("Agent")

        # Action: Check Agent Diagnostics
        status_action = agent_menu.addAction("🔍 Agent Diagnostics & Status")
        status_action.setShortcut("Ctrl+D")
        status_action.triggered.connect(self.show_agent_status_dialog)

    def show_agent_status_dialog(self):
        """Fetches status from main window and displays a diagnostic popup."""
        status_info = {
            "status": "idle",
            "current_step": 0,
            "last_action": "No agent running",
            "error": None,
        }

        # Query main window if available
        if (
            self.main_window
            and hasattr(self.main_window, "get_agent_status")
            and callable(self.main_window.get_agent_status)
        ):
            status_info = self.main_window.get_agent_status()

        # Format diagnostic message
        status_str = status_info.get("status", "unknown").upper()
        step_count = status_info.get("current_step", 0)
        last_act = status_info.get("last_action", "N/A")
        err = status_info.get("error", None)

        msg = (
            f"<b>Agent Status:</b> {status_str}<br>"
            f"<b>Steps Executed:</b> {step_count}<br>"
            f"<b>Last Action:</b> {last_act}"
        )

        if err:
            msg += f"<br><br><span style='color: red;'><b>Error:</b> {err}</span>"

        # Display popup dialog
        QMessageBox.information(self.main_window, "Agent Diagnostics", msg)

# chat_formatter.py
import html
import re
from datetime import datetime


class ChatMessageFormatter:
    """Single-responsibility class for parsing raw AI/User text and rendering

    industrial-grade CSS styled HTML bubbles for PySide6 QTextEdit display.
    """

    # Industrial Theme Palette
    COLOR_USER_BG = "#f4f4f5"
    COLOR_USER_BORDER = "#e4e4e7"
    COLOR_ASSISTANT_BG = "#f0f7ff"
    COLOR_ASSISTANT_BORDER = "#bfdbfe"
    COLOR_SYSTEM_BG = "#fefce8"
    COLOR_SYSTEM_BORDER = "#fef08a"
    COLOR_AGENT_BG = "#faf5ff"
    COLOR_AGENT_BORDER = "#e9d5ff"
    COLOR_ERROR_BG = "#fef2f2"
    COLOR_ERROR_BORDER = "#fecaca"

    @classmethod
    def format_message(cls, role: str, raw_text: str) -> str:
        """Formats raw input text into a styled HTML card based on the role.

        Supported roles: 'user', 'assistant', 'system', 'agent', 'error'
        """
        timestamp = datetime.now().strftime("%H:%M:%S")
        processed_body = cls._parse_markdown(raw_text)

        if role == "user":
            title = "USER"
            badge_color = "#18181b"
            bg_color = cls.COLOR_USER_BG
            border_color = cls.COLOR_USER_BORDER
        elif role == "assistant":
            title = "GEMINI AI"
            badge_color = "#2563eb"
            bg_color = cls.COLOR_ASSISTANT_BG
            border_color = cls.COLOR_ASSISTANT_BORDER
        elif role == "agent":
            title = "BROWSER AGENT"
            badge_color = "#9333ea"
            bg_color = cls.COLOR_AGENT_BG
            border_color = cls.COLOR_AGENT_BORDER
        elif role == "system":
            title = "SYSTEM"
            badge_color = "#ca8a04"
            bg_color = cls.COLOR_SYSTEM_BG
            border_color = cls.COLOR_SYSTEM_BORDER
        elif role == "error":
            title = "SYSTEM ERROR"
            badge_color = "#dc2626"
            bg_color = cls.COLOR_ERROR_BG
            border_color = cls.COLOR_ERROR_BORDER
        else:
            title = role.upper()
            badge_color = "#52525b"
            bg_color = "#ffffff"
            border_color = "#e4e4e7"

        formatted_html = f"""
        <div style="margin-bottom: 12px; background-color: {bg_color}; border: 1px solid {border_color}; border-radius: 6px; padding: 10px 12px; font-family: -apple-system, BlinkMacSystemFont, 'Segoe UI', Roboto, Helvetica, Arial, sans-serif;">
            <div style="margin-bottom: 6px; font-size: 11px;">
                <span style="background-color: {badge_color}; color: #ffffff; font-weight: bold; padding: 2px 6px; border-radius: 3px; letter-spacing: 0.5px;">{title}</span>
                <span style="color: #71717a; margin-left: 8px;">{timestamp}</span>
            </div>
            <div style="font-size: 13px; line-height: 1.5; color: #09090b;">
                {processed_body}
            </div>
        </div>
        """
        return formatted_html

    @classmethod
    def _parse_markdown(cls, text: str) -> str:
        """Lightweight Markdown and Code syntax parser to convert LLM output into

        rich HTML display format.
        """
        if not text:
            return ""

        # Escape HTML tags to prevent XSS / render corruption
        escaped_text = html.escape(text)

        # 1. Parse multi-line code blocks ```lang ... ```
        code_block_pattern = re.compile(
            r"```(?:(\w+)\n)?(.*?)```", re.DOTALL
        )

        def replace_code_block(match):
            lang = match.group(1) or "code"
            code_content = match.group(2).strip()
            return f"""
            <div style="margin: 8px 0; background-color: #1e1e2e; border-radius: 4px; border: 1px solid #313244; overflow: hidden;">
                <div style="background-color: #181825; padding: 3px 8px; font-size: 10px; color: #a6adc8; font-family: monospace; font-weight: bold; border-bottom: 1px solid #313244;">
                    {lang.upper()}
                </div>
                <pre style="margin: 0; padding: 8px; color: #cdd6f4; font-family: 'Consolas', 'Courier New', monospace; font-size: 12px; white-space: pre-wrap;">{code_content}</pre>
            </div>
            """

        escaped_text = code_block_pattern.sub(replace_code_block, escaped_text)

        # 2. Parse inline code `code`
        inline_code_pattern = re.compile(r"`([^`]+)`")
        escaped_text = inline_code_pattern.sub(
            r'<code style="background-color: #e4e4e7; color: #09090b; padding: 2px 4px; border-radius: 3px; font-family: monospace; font-size: 12px;">\1</code>',
            escaped_text,
        )

        # 3. Bold **text**
        escaped_text = re.sub(
            r"\*\*(.*?)\*\*", r"<strong>\1</strong>", escaped_text
        )

        # 4. Italic *text*
        escaped_text = re.sub(r"\*(.*?)\*", r"<em>\1</em>", escaped_text)

        # 5. Convert linebreaks outside code blocks
        escaped_text = escaped_text.replace("\n", "<br>")

        return escaped_text

import json
import os
from google import genai
from google.genai import types
from PySide6.QtCore import Qt, QThread, Signal, Slot
from PySide6.QtWidgets import (
    QComboBox,
    QFrame,

    QLineEdit,

from chat_formatter import ChatMessageFormatter
from PySide6.QtGui import QTextCursor

class GeminiWorker(QThread):
    chat_chunk = Signal(str)
    chat_finished = Signal()
    task_response = Signal(str)
    error_occurred = Signal(str)

    def __init__(self, prompt, api_key, chat_session=None):
        super().__init__()
        self.prompt = prompt
        self.api_key = api_key
        self.chat_session = chat_session

    def run(self):
        try:
            client = genai.Client(api_key=self.api_key)

            # System instruction enforcing JSON intent classification
            system_instruction = (
                "You are an intent classification system for an AI assistant. "
                "Classify the user prompt as either 'BROWSER_TASK' or 'CHAT'.\n"
                "- Use 'BROWSER_TASK' if the prompt involves web navigation, searching online, clicking, scraping, opening URLs, or interacting with web applications.\n"
                "- Use 'CHAT' if the user is asking general questions, chatting, coding, or summarizing without needing live web actions.\n"
                "Return JSON matching the schema."
            )

            # Load JSON schema structure
            schema_path = os.path.join(
                os.path.dirname(__file__), "intent_schema.json"
            )
            response_schema = None
            if os.path.exists(schema_path):
                with open(schema_path, "r") as f:
                    response_schema = json.load(f)

            config = types.GenerateContentConfig(
                system_instruction=system_instruction,
                response_mime_type="application/json",
            )

            # Fast single turn call to classify intent
            classification_prompt = f"Analyze intent for: '{self.prompt}'"
            response = client.models.generate_content(
                model="gemini-3.5-flash-lite",
                contents=classification_prompt,
                config=config,
            )

            data = json.loads(response.text)
            intent = data.get("intent", "CHAT")
            task_desc = data.get("task_description", self.prompt)

            if intent == "BROWSER_TASK":
                self.task_response.emit(task_desc)
            else:
                # Handle general chat session with streaming
                if self.chat_session is None:
                    chat = client.chats.create(model="gemini-3.5-flash-lite")
                else:
                    chat = self.chat_session

                response_stream = chat.send_message_stream(self.prompt)
                for chunk in response_stream:
                    if chunk.text:
                        self.chat_chunk.emit(chunk.text)
                self.chat_finished.emit()

        except Exception as e:
            self.error_occurred.emit(str(e))

# agent_engine.py
import asyncio
import os
from PySide6.QtCore import QThread, Signal
from browser_use import Agent, Browser, ChatGoogle

from agent_state_manager import AgentStateManager


class AgentExecutionWorker(QThread):
    step_updated = Signal(str)
    task_finished = Signal(str)
    error_occurred = Signal(str)

    def __init__(
        self,
        task_description,
        api_key,
        headless=False,
        cdp_url="http://127.0.0.1:9222",
    ):
        super().__init__()
        self.task_description = task_description
        self.api_key = api_key
        self.headless = headless
        self.cdp_url = cdp_url

        # Delegate state tracking to independent state manager
        self.state_manager = AgentStateManager()

    def get_status(self) -> dict:
        """API/UI Endpoint method to query current agent state at any time."""
        return self.state_manager.get_status()

    async def _on_step_end_callback(self, agent: Agent):
        """Lifecycle hook executed after every single agent step."""
        formatted_step_msg = self.state_manager.record_step(agent)
        self.step_updated.emit(formatted_step_msg)

    def run(self):
        # Create a clean asyncio event loop for this QThread
        loop = asyncio.new_event_loop()
        asyncio.set_event_loop(loop)
        try:
            loop.run_until_complete(self._execute_agent_task())
        except Exception as e:
            err_msg = str(e)
            self.state_manager.mark_failed(err_msg)
            self.error_occurred.emit(err_msg)
        finally:
            loop.close()

    async def _execute_agent_task(self):
        try:
            self.state_manager.start_task()

            # Pass API key into environment for ChatGoogle
            os.environ["GOOGLE_API_KEY"] = self.api_key

            # Initialize LLM with high-throughput free-tier model
            llm = ChatGoogle(model="gemini-3.5-flash-lite")

            # Connect to existing embedded Qt WebEngine browser via CDP if provided
            if self.cdp_url:
                browser = Browser(cdp_url=self.cdp_url)
            else:
                browser = Browser(headless=self.headless)

            # Instantiate Agent with fail-safes and step hooks
            agent = Agent(
                task=self.task_description,
                llm=llm,
                browser=browser,
                llm_timeout=30,  # Max 30s for LLM API response
                step_timeout=60,  # Max 60s per DOM action
                max_failures=3,  # Max consecutive retries on error
                final_response_after_failure=True,  # Return best-effort summary if max_failures reached
            )

            self.step_updated.emit(
                f"🚀 Initializing Agent for task: '{self.task_description}'"
            )

            # Run loop with step monitoring callback
            result = await agent.run(on_step_end=self._on_step_end_callback)

            self.state_manager.mark_completed()
            self.task_finished.emit(str(result))

        except asyncio.TimeoutError:
            err_msg = "Agent execution timed out while waiting for model or browser response."
            self.state_manager.mark_failed(err_msg)
            self.error_occurred.emit(err_msg)

        except Exception as e:
            err_msg = f"Agent Execution Failure: {str(e)}"
            self.state_manager.mark_failed(err_msg)
            self.error_occurred.emit(err_msg)

# agent_response_parser.py
import html
import re
from typing import Dict, List, Tuple


class AgentResponseParser:
    """Predictive parsing engine for browser-use agent outputs."""

    @classmethod
    def parse_and_format(cls, raw_result_str: str) -> str:
        """Parses raw AgentHistoryList string or plain text into formatted HTML

        with cascading styles and collapsible step logs.
        """
        final_summary, extracted_steps = cls._extract_semantics(raw_result_str)

        # 1. Format the main output card
        formatted_summary = cls._format_summary_card(final_summary)

        # 2. Format step telemetry into a collapsible HTML block
        formatted_steps = cls._format_collapsible_steps(extracted_steps)

        return f"{formatted_summary}\n{formatted_steps}"

    @classmethod
    def _extract_semantics(
        cls, raw_str: str
    ) -> Tuple[str, List[Dict[str, str]]]:
        """Predictively extracts final outcome and intermediate execution steps."""
        steps = []
        final_summary = ""

        # Regex patterns to capture intermediate ActionResults
        action_pattern = re.compile(
            r"ActionResult\([^)]*?extracted_content=['\"](.*?)['\"]", re.DOTALL
        )
        for match in action_pattern.finditer(raw_str):
            try:
                content = match.group(1).encode("utf-8").decode("unicode_escape")
            except Exception:
                content = match.group(1)

            if content.strip() and not content.startswith("Match:"):
                clean_content = (
                    content[:120] + "..." if len(content) > 120 else content
                )
                steps.append(
                    {"type": "Action / DOM Step", "details": clean_content}
                )

        # Pattern for final text inside done(...)
        done_pattern = re.compile(
            r"done=['\"]?\{'text':\s*['\"](.*?)['\"]", re.DOTALL
        )
        done_match = done_pattern.search(raw_str)

        if done_match:
            try:
                final_summary = (
                    done_match.group(1)
                    .encode("utf-8")
                    .decode("unicode_escape")
                )
            except Exception:
                final_summary = done_match.group(1)
        else:
            # Fallback for extracted content near end of response
            summary_pattern = re.compile(
                r"extracted_content=['\"](Match:[^'\"]+)['\"]", re.DOTALL
            )
            summary_match = summary_pattern.search(raw_str)
            if summary_match:
                final_summary = summary_match.group(1)
            else:
                final_summary = raw_str

        return final_summary, steps

    @classmethod
    def _format_summary_card(cls, text: str) -> str:
        """Renders the main agent output inside a visual CSS container."""
        if not text:
            text = "Task executed successfully with no returned text."

        # Escape HTML characters first to prevent XSS/broken layout
        text_escaped = html.escape(text)

        # Convert simple markdown to HTML inside summary card
        # Bold
        text_escaped = re.sub(
            r"\*\*(.*?)\*\*", r"<strong>\1</strong>", text_escaped
        )

        # Bullets (- Item)
        lines = text_escaped.split("\n")
        formatted_lines = []
        for line in lines:
            line_str = line.strip()
            if line_str.startswith("- ") or line_str.startswith("* "):
                formatted_lines.append(
                    f"<li style='margin-left: 15px;'>{line_str[2:]}</li>"
                )
            elif line_str:
                formatted_lines.append(f"<div>{line_str}</div>")

        processed_body = "".join(formatted_lines)

        return f"""
        <div style="background-color: #f8fafc; border-left: 4px solid #16a34a; padding: 10px 12px; margin-top: 6px; margin-bottom: 8px; border-radius: 4px;">
            <div style="font-weight: bold; color: #1e293b; font-size: 13px; margin-bottom: 6px;">📌 Execution Result</div>
            <div style="font-size: 13px; color: #334155; line-height: 1.5;">{processed_body}</div>
        </div>
        """

    @classmethod
    def _format_collapsible_steps(cls, steps: List[Dict[str, str]]) -> str:
        """Generates an HTML details/summary collapsible dropdown for agent steps."""
        if not steps:
            return ""

        step_items_html = ""
        for idx, step in enumerate(steps, 1):
            clean_details = html.escape(step["details"])
            step_items_html += f"""
            <li style="margin-bottom: 4px;">
                <strong style="color: #64748b;">Step {idx}:</strong> <code>{clean_details}</code>
            </li>
            """

        return f"""
        <details style="margin-top: 6px; background-color: #f1f5f9; border: 1px solid #cbd5e1; border-radius: 4px; padding: 6px 10px; font-size: 12px;">
            <summary style="font-weight: bold; color: #475569; cursor: pointer; user-select: none;">
                🔍 View Agent Execution Steps ({len(steps)} actions)
            </summary>
            <ul style="margin: 8px 0 4px 18px; padding: 0; color: #334155; font-family: monospace;">
                {step_items_html}
            </ul>
        </details>
        """

# agent_state_manager.py
from typing import Any, Dict, Optional


class AgentStateManager:
    """Standalone, testable state monitor for browser agent execution."""

    def __init__(self):
        self._state: Dict[str, Any] = {
            "status": "idle",  # "idle", "running", "completed", "failed"
            "current_step": 0,
            "last_action": "Not started",
            "error": None,
        }

    def start_task(self) -> None:
        """Resets and transitions state to running."""
        self._state["status"] = "running"
        self._state["current_step"] = 0
        self._state["last_action"] = "Initializing..."
        self._state["error"] = None

    def record_step(self, agent_instance: Optional[Any] = None) -> str:
        """Parses the latest action from the agent history and increments step count."""
        self._state["current_step"] += 1
        last_action_desc = "Processing step..."

        if agent_instance and hasattr(agent_instance, "history"):
            history = agent_instance.history
            if history and hasattr(history, "model_actions") and history.model_actions():
                last_action_desc = str(history.model_actions()[-1])

        self._state["last_action"] = last_action_desc
        return f"Step {self._state['current_step']}: {last_action_desc[:80]}..."

    def mark_completed(self) -> None:
        """Transitions state to completed."""
        self._state["status"] = "completed"

    def mark_failed(self, error_message: str) -> None:
        """Transitions state to failed with error details."""
        self._state["status"] = "failed"
        self._state["error"] = error_message

    def get_status(self) -> Dict[str, Any]:
        """Returns a copy of the current state snapshot for UI / API endpoints."""
        return self._state.copy()

import html
import re
from datetime import datetime


class ChatMessageFormatter:
    """Single-responsibility class for parsing raw AI/User text and rendering

    industrial-grade CSS styled HTML bubbles for PySide6 QTextEdit display.
    """

    # Industrial Theme Palette
    COLOR_USER_BG = "#f4f4f5"
    COLOR_USER_BORDER = "#e4e4e7"
    COLOR_ASSISTANT_BG = "#f0f7ff"
    COLOR_ASSISTANT_BORDER = "#bfdbfe"
    COLOR_SYSTEM_BG = "#fefce8"
    COLOR_SYSTEM_BORDER = "#fef08a"
    COLOR_AGENT_BG = "#faf5ff"
    COLOR_AGENT_BORDER = "#e9d5ff"
    COLOR_ERROR_BG = "#fef2f2"
    COLOR_ERROR_BORDER = "#fecaca"

    @classmethod
    def format_message(cls, role: str, raw_text: str) -> str:
        """Formats raw input text into a styled HTML card based on the role.

        Supported roles: 'user', 'assistant', 'system', 'agent', 'error'
        """
        timestamp = datetime.now().strftime("%H:%M:%S")
        processed_body = cls._parse_markdown(raw_text)

        if role == "user":
            title = "USER"
            badge_color = "#18181b"
            bg_color = cls.COLOR_USER_BG
            border_color = cls.COLOR_USER_BORDER
        elif role == "assistant":
            title = "GEMINI AI"
            badge_color = "#2563eb"
            bg_color = cls.COLOR_ASSISTANT_BG
            border_color = cls.COLOR_ASSISTANT_BORDER
        elif role == "agent":
            title = "BROWSER AGENT"
            badge_color = "#9333ea"
            bg_color = cls.COLOR_AGENT_BG
            border_color = cls.COLOR_AGENT_BORDER
        elif role == "system":
            title = "SYSTEM"
            badge_color = "#ca8a04"
            bg_color = cls.COLOR_SYSTEM_BG
            border_color = cls.COLOR_SYSTEM_BORDER
        elif role == "error":
            title = "SYSTEM ERROR"
            badge_color = "#dc2626"
            bg_color = cls.COLOR_ERROR_BG
            border_color = cls.COLOR_ERROR_BORDER
        else:
            title = role.upper()
            badge_color = "#52525b"
            bg_color = "#ffffff"
            border_color = "#e4e4e7"

        formatted_html = f"""
        <div style="margin-bottom: 12px; background-color: {bg_color}; border: 1px solid {border_color}; border-radius: 6px; padding: 10px 12px; font-family: -apple-system, BlinkMacSystemFont, 'Segoe UI', Roboto, Helvetica, Arial, sans-serif;">
            <div style="margin-bottom: 6px; font-size: 11px;">
                <span style="background-color: {badge_color}; color: #ffffff; font-weight: bold; padding: 2px 6px; border-radius: 3px; letter-spacing: 0.5px;">{title}</span>
                <span style="color: #71717a; margin-left: 8px;">{timestamp}</span>
            </div>
            <div style="font-size: 13px; line-height: 1.5; color: #09090b;">
                {processed_body}
            </div>
        </div>
        """
        return formatted_html

    @classmethod
    def _parse_markdown(cls, text: str) -> str:
        """Lightweight Markdown and Code syntax parser to convert LLM output into

        rich HTML display format.
        """
        if not text:
            return ""

        # Escape HTML tags to prevent XSS / render corruption
        escaped_text = html.escape(text)

        # 1. Parse multi-line code blocks ```lang ... ```
        code_block_pattern = re.compile(
            r"```(?:(\w+)\n)?(.*?)```", re.DOTALL
        )

        def replace_code_block(match):
            lang = match.group(1) or "code"
            code_content = match.group(2).strip()
            return f"""
            <div style="margin: 8px 0; background-color: #1e1e2e; border-radius: 4px; border: 1px solid #313244; overflow: hidden;">
                <div style="background-color: #181825; padding: 3px 8px; font-size: 10px; color: #a6adc8; font-family: monospace; font-weight: bold; border-bottom: 1px solid #313244;">
                    {lang.upper()}
                </div>
                <pre style="margin: 0; padding: 8px; color: #cdd6f4; font-family: 'Consolas', 'Courier New', monospace; font-size: 12px; white-space: pre-wrap;">{code_content}</pre>
            </div>
            """

        escaped_text = code_block_pattern.sub(replace_code_block, escaped_text)

        # 2. Parse inline code `code`
        inline_code_pattern = re.compile(r"`([^`]+)`")
        escaped_text = inline_code_pattern.sub(
            r'<code style="background-color: #e4e4e7; color: #09090b; padding: 2px 4px; border-radius: 3px; font-family: monospace; font-size: 12px;">\1</code>',
            escaped_text,
        )

        # 3. Bold **text**
        escaped_text = re.sub(
            r"\*\*(.*?)\*\*", r"<strong>\1</strong>", escaped_text
        )

        # 4. Italic *text*
        escaped_text = re.sub(r"\*(.*?)\*", r"<em>\1</em>", escaped_text)

        # 5. Convert linebreaks outside code blocks
        escaped_text = escaped_text.replace("\n", "<br>")

        return escaped_text

class GeminiWorker(QThread):
    chat_chunk = Signal(str)
    chat_finished = Signal()
    task_response = Signal(str)
    error_occurred = Signal(str)

    def __init__(self, prompt, api_key, chat_session=None):
        super().__init__()
        self.prompt = prompt
        self.api_key = api_key
        self.chat_session = chat_session

    def run(self):
        try:
            client = genai.Client(api_key=self.api_key)

            # System instruction enforcing JSON intent classification
            system_instruction = (
                "You are an intent classification system for an AI assistant. "
                "Classify the user prompt as either 'BROWSER_TASK' or 'CHAT'.\n"
                "- Use 'BROWSER_TASK' if the prompt involves web navigation, searching online, clicking, scraping, opening URLs, or interacting with web applications.\n"
                "- Use 'CHAT' if the user is asking general questions, chatting, coding, or summarizing without needing live web actions.\n"
                "Return JSON matching the schema."
            )

            # Load JSON schema structure
            schema_path = os.path.join(
                os.path.dirname(__file__), "intent_schema.json"
            )
            response_schema = None
            if os.path.exists(schema_path):
                with open(schema_path, "r") as f:
                    response_schema = json.load(f)

            config = types.GenerateContentConfig(
                system_instruction=system_instruction,
                response_mime_type="application/json",
            )

            # Fast single turn call to classify intent
            classification_prompt = f"Analyze intent for: '{self.prompt}'"
            response = client.models.generate_content(
                model="gemini-3.5-flash-lite",
                contents=classification_prompt,
                config=config,
            )

            data = json.loads(response.text)
            intent = data.get("intent", "CHAT")
            task_desc = data.get("task_description", self.prompt)

            if intent == "BROWSER_TASK":
                self.task_response.emit(task_desc)
            else:
                # Handle general chat session with streaming
                if self.chat_session is None:
                    chat = client.chats.create(model="gemini-3.5-flash-lite")
                else:
                    chat = self.chat_session

                response_stream = chat.send_message_stream(self.prompt)
                for chunk in response_stream:
                    if chunk.text:
                        self.chat_chunk.emit(chunk.text)
                self.chat_finished.emit()

        except Exception as e:
            self.error_occurred.emit(str(e))

            self.is_first_chunk = False

            cursor.movePosition(QTextCursor.MoveOperation.End)

        self.chat_display.append(err_html)
        self._reset_input_state()

    def _reset_input_state(self):
        self.input_field.setEnabled(True)
        self.send_btn.setEnabled(True)
        self.input_field.setFocus()

# task_graph.py
from dataclasses import dataclass, field
from typing import List, Optional

@dataclass
class TaskStep:
    step_id: int
    description: str
    action_type: str  # e.g., 'navigate', 'click', 'type', 'extract'
    status: str = "PENDING"  # PENDING, EXECUTING, COMPLETED, FAILED
    risk_score: float = 0.0

class TaskGraph:
    def __init__(self, task_id: str, prompt: str):
        self.task_id = task_id
        self.prompt = prompt
        self.steps: List[TaskStep] = []
        self.current_step_index: int = 0

    def add_step(self, description: str, action_type: str):
        step_id = len(self.steps) + 1
        self.steps.append(TaskStep(step_id, description, action_type))

    def get_current_step(self) -> Optional[TaskStep]:
        if self.current_step_index < len(self.steps):
            return self.steps[self.current_step_index]
        return None

    def advance(self):
        self.current_step_index += 1
