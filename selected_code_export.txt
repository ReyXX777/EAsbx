from __future__ import annotations

from typing import Any, Mapping

from PySide6.QtCore import QObject, QThread, Signal, Slot

from .observation_thread import ObservationWorker


class ObservationBridge(QObject):
    """Thread-safe Qt bridge for the observation worker.

    UI code communicates only with this object.

    The bridge owns the worker thread and forwards requests to the
    ObservationWorker using Qt's queued signal/slot mechanism.
    """

    task_started = Signal(str)
    observation_updated = Signal(object)
    task_finished = Signal(object)
    error = Signal(str)

    request_start = Signal(str, str, object)
    request_discrete = Signal(str, object)
    request_continuous = Signal(str, object)
    request_event = Signal(str, object)
    request_snapshot = Signal()
    request_finish = Signal()
    request_shutdown = Signal()

    def __init__(
        self,
        storage_root: str,
        parent: QObject | None = None,
    ) -> None:
        super().__init__(parent)

        self._closed = False

        self._thread = QThread(self)

        self._worker = ObservationWorker(
            storage_root,
        )

        self._worker.moveToThread(
            self._thread,
        )

        # --------------------------------------------------------------
        # Requests: UI -> worker
        # --------------------------------------------------------------

        self.request_start.connect(
            self._worker.start_task,
        )

        self.request_discrete.connect(
            self._worker.record_discrete,
        )

        self.request_continuous.connect(
            self._worker.record_continuous,
        )

        self.request_event.connect(
            self._worker.record_event,
        )

        self.request_snapshot.connect(
            self._worker.publish_current_state,
        )

        self.request_finish.connect(
            self._worker.finish_task,
        )

        self.request_shutdown.connect(
            self._worker.shutdown,
        )

        # --------------------------------------------------------------
        # Results: worker -> UI
        # --------------------------------------------------------------

        self._worker.task_started.connect(
            self.task_started,
        )

        self._worker.observation_updated.connect(
            self.observation_updated,
        )

        self._worker.task_finished.connect(
            self.task_finished,
        )

        self._worker.worker_error.connect(
            self.error,
        )

        self._thread.start()

    # ------------------------------------------------------------------
    # Public commands
    # ------------------------------------------------------------------

    @Slot(str, str, object)
    def start_task(
        self,
        family_key: str,
        task_key: str = "",
        metadata: Mapping[str, Any] | None = None,
    ) -> None:
        """Request creation of a new observation session."""

        if self._closed:
            return

        self.request_start.emit(
            family_key,
            task_key,
            metadata,
        )

    @Slot(str, object)
    def record_discrete(
        self,
        name: str,
        value: Any = None,
    ) -> None:
        """Request a discrete observation."""

        if self._closed:
            return

        self.request_discrete.emit(
            name,
            value,
        )

    @Slot(str, object)
    def record_continuous(
        self,
        name: str,
        value: Any = None,
    ) -> None:
        """Request a continuous observation."""

        if self._closed:
            return

        self.request_continuous.emit(
            name,
            value,
        )

    @Slot(str, object)
    def record_event(
        self,
        event_type: str,
        payload: Any = None,
    ) -> None:
        """Request an event observation."""

        if self._closed:
            return

        self.request_event.emit(
            event_type,
            payload,
        )

    def request_current_snapshot(self) -> None:
        """Request publication of the current live snapshot."""

        if self._closed:
            return

        self.request_snapshot.emit()

    def finish_task(self) -> None:
        """Request finalization of the active observation session."""

        if self._closed:
            return

        self.request_finish.emit()

    # ------------------------------------------------------------------
    # Lifecycle
    # ------------------------------------------------------------------

    def close(self) -> None:
        """Stop the worker thread safely and release Qt resources."""

        if self._closed:
            return

        self._closed = True

        if not self._thread.isRunning():
            return

        self.request_shutdown.emit()
        self._thread.quit()

        if not self._thread.wait(3000):
            self._thread.terminate()
            self._thread.wait(1000)

    def __del__(self) -> None:
        try:
            self.close()
        except Exception:
            pass

from __future__ import annotations

from typing import Any, Mapping

from PySide6.QtCore import QObject, Signal, Slot

from ..models import ObservationSnapshot
from ..session import TaskObservationEngine, TaskObservationSession


class ObservationWorker(QObject):
    """Owns live observation state on the dedicated observation thread.

    The worker is the only Qt-side object allowed to mutate the active
    TaskObservationSession.

    The presentation layer communicates with this object through queued
    signals exposed by ObservationBridge.

    Responsibilities:

    - create live observation sessions
    - mutate collectors belonging to the active session
    - publish current snapshots
    - finish sessions
    - release the active session

    Persistence and feature derivation remain owned by the observation
    engine/session layer.
    """

    task_started = Signal(str)
    observation_updated = Signal(object)
    task_finished = Signal(object)
    worker_error = Signal(str)

    EVENT_SOURCE = "qt_observation_worker"

    def __init__(
        self,
        storage_root: str,
        parent: QObject | None = None,
    ) -> None:
        super().__init__(parent)

        self._engine = TaskObservationEngine(
            storage_root=storage_root,
        )

        self._session: TaskObservationSession | None = None

    # ------------------------------------------------------------------
    # Session lifecycle
    # ------------------------------------------------------------------

    @Slot(str, str, object)
    def start_task(
        self,
        family_key: str,
        task_key: str = "",
        metadata: Mapping[str, Any] | None = None,
    ) -> None:
        """Start a new observation session."""

        try:
            if self._session is not None:
                self.worker_error.emit(
                    "Cannot start a new task while another task is active."
                )
                return

            family_key = (family_key or "").strip()

            if not family_key:
                self.worker_error.emit(
                    "Cannot start an observation without a task family."
                )
                return

            task_key = (task_key or "").strip()

            self._session = self._engine.start(
                family_key,
                task_key=task_key,
                metadata=metadata,
            )

            self.task_started.emit(family_key)

            self.observation_updated.emit(
                self._session.snapshot(),
            )

        except Exception as exc:
            self.worker_error.emit(f"Task start error: {type(exc).__name__}: {exc}")

    @Slot()
    def finish_task(self) -> None:
        """Finish the active observation session.

        TaskObservationSession.finish() is deliberately responsible for
        producing the final snapshot, including any configured feature
        derivation/persistence performed by the observation engine.
        """

        session = self._session

        if session is None:
            self.worker_error.emit("No active observation session.")
            return

        try:
            snapshot = session.finish()

            self._session = None

            self.task_finished.emit(
                snapshot,
            )

            self.observation_updated.emit(
                snapshot,
            )

        except Exception as exc:
            self.worker_error.emit(
                f"Task finalization error: {type(exc).__name__}: {exc}"
            )

    # ------------------------------------------------------------------
    # Discrete observations
    # ------------------------------------------------------------------

    @Slot(str, object)
    def record_discrete(
        self,
        name: str,
        value: Any = None,
    ) -> None:
        """Record a discrete observation on the active session."""

        session = self._session

        if session is None:
            self.worker_error.emit("No active observation session.")
            return

        try:
            session.discrete.record(
                name,
                value,
            )

        except Exception as exc:
            self.worker_error.emit(
                "Discrete observation error: " f"{type(exc).__name__}: {exc}"
            )

    # ------------------------------------------------------------------
    # Continuous observations
    # ------------------------------------------------------------------

    @Slot(str, object)
    def record_continuous(
        self,
        name: str,
        value: Any = None,
    ) -> None:
        """Record a continuous observation on the active session."""

        session = self._session

        if session is None:
            self.worker_error.emit("No active observation session.")
            return

        try:
            session.continuous.record(
                name,
                value,
            )

        except Exception as exc:
            self.worker_error.emit(
                "Continuous observation error: " f"{type(exc).__name__}: {exc}"
            )

    # ------------------------------------------------------------------
    # Events
    # ------------------------------------------------------------------

    @Slot(str, object)
    def record_event(
        self,
        event_type: str,
        payload: Any = None,
    ) -> None:
        """Record an event and its canonical trace representation."""

        session = self._session

        if session is None:
            self.worker_error.emit("No active observation session.")
            return

        try:
            normalized_payload = (
                {}
                if payload is None
                else (
                    dict(payload)
                    if isinstance(payload, Mapping)
                    else {"value": payload}
                )
            )

            session.events.event(
                event_type,
                source=self.EVENT_SOURCE,
                payload=normalized_payload,
            )

            # Tool lifecycle events participate in both the event stream
            # and the canonical trace stream.
            if event_type in {
                "tool_started",
                "tool_completed",
                "tool_failed",
            }:
                tool_name = normalized_payload.get("tool_name")

                if tool_name:
                    path = normalized_payload.get("path")

                    trace_discrete = {
                        key: value
                        for key, value in normalized_payload.items()
                        if key not in {"tool_name", "path"}
                    }

                    trace_discrete["tool_name"] = str(tool_name)

                    session.events.trace(
                        event_type,
                        action=str(tool_name),
                        path=(str(path) if path is not None else None),
                        discrete=trace_discrete,
                    )

        except Exception as exc:
            self.worker_error.emit(
                "Event observation error: " f"{type(exc).__name__}: {exc}"
            )

    # ------------------------------------------------------------------
    # Snapshot publication
    # ------------------------------------------------------------------

    @Slot()
    def publish_current_state(self) -> None:
        """Publish the current raw observation snapshot."""

        session = self._session

        if session is None:
            return

        try:
            snapshot = session.snapshot()

            self.observation_updated.emit(
                snapshot,
            )

        except Exception as exc:
            self.worker_error.emit(f"Snapshot error: {type(exc).__name__}: {exc}")

    # ------------------------------------------------------------------
    # Shutdown
    # ------------------------------------------------------------------

    @Slot()
    def shutdown(self) -> None:
        """Release the active session reference.

        A live unfinished session is deliberately not finalized here.
        Normal persistence occurs through finish_task().
        """

        self._session = None

from __future__ import annotations

import math
from typing import Iterable

from PySide6.QtCore import QPointF, Qt
from PySide6.QtGui import QPainter
from PySide6.QtWidgets import QWidget


class ObservationPlot(QWidget):
    """Small dependency-free Qt plot for continuous observations.

    The plot is intentionally independent of:

    - TaskObservationSession
    - TaskObservationEngine
    - SessionStore
    - ReplayAPI
    - collector implementations

    It only receives numeric values and renders them.
    """

    def __init__(
        self,
        parent: QWidget | None = None,
    ) -> None:
        super().__init__(parent)

        self.setMinimumHeight(220)

        self._values: list[float] = []
        self._title = "Continuous Feature"

    def set_data(
        self,
        values: Iterable[float],
        title: str = "Continuous Feature",
    ) -> None:
        """Replace the plotted values."""

        cleaned: list[float] = []

        for value in values:
            try:
                numeric = float(value)
            except (TypeError, ValueError):
                continue

            if not math.isfinite(numeric):
                continue

            cleaned.append(numeric)

        self._values = cleaned
        self._title = str(title)

        self.update()

    def clear(self) -> None:
        """Clear the current plot."""

        self._values.clear()
        self.update()

    def paintEvent(self, _event) -> None:
        painter = QPainter(self)

        try:
            painter.setRenderHint(
                QPainter.Antialiasing,
                True,
            )

            rect = self.rect().adjusted(
                45,
                25,
                -20,
                -35,
            )

            painter.drawText(
                10,
                18,
                self._title,
            )

            if not self._values:
                painter.drawText(
                    rect,
                    Qt.AlignCenter,
                    "No numeric observations.",
                )
                return

            minimum = min(self._values)
            maximum = max(self._values)

            if math.isclose(
                minimum,
                maximum,
            ):
                padding = max(
                    abs(minimum) * 0.05,
                    1.0,
                )

                minimum -= padding
                maximum += padding

            painter.drawRect(
                rect,
            )

            point_count = len(self._values)

            points: list[QPointF] = []

            for index, value in enumerate(self._values):
                if point_count == 1:
                    x = float(rect.center().x())
                else:
                    x = rect.left() + (index / (point_count - 1)) * rect.width()

                normalized = (value - minimum) / (maximum - minimum)

                y = rect.bottom() - normalized * rect.height()

                points.append(
                    QPointF(
                        x,
                        y,
                    )
                )

            if len(points) >= 2:
                painter.drawPolyline(
                    points,
                )
            else:
                painter.drawEllipse(
                    points[0],
                    3,
                    3,
                )

            painter.drawText(
                5,
                rect.top(),
                f"{maximum:.4g}",
            )

            painter.drawText(
                5,
                rect.bottom(),
                f"{minimum:.4g}",
            )

            painter.drawText(
                rect.left(),
                rect.bottom() + 25,
                "start",
            )

            painter.drawText(
                rect.right() - 35,
                rect.bottom() + 25,
                "end",
            )

        finally:
            painter.end()

        self.status_label.setText(f"Collecting: {family}")

    def _on_task_finished(
        self,
        snapshot: object,
    ) -> None:
        if not isinstance(
            snapshot,
            ObservationSnapshot,
        ):
            self._on_error("Worker returned an invalid observation snapshot.")
            return

        self.status_label.setText("Observation completed.")

        self._display_snapshot(
            snapshot,
        )

        self._set_running_state(
            False,
        )

        self.refresh_history()

        self,

        self.status_label.setText(f"Observation error: {message}")

        self,

        self.start_button.setEnabled(

        self,

        if not isinstance(

        identity = snapshot.identity

        self.session_id_label.setText(

            str(identity.family_key),

            self._safe_text(snapshot.finished_at),
        )

        self._display_discrete(
            snapshot.discrete,
        )

        self._display_continuous(
            snapshot.continuous,
        )

        self._display_events(
            snapshot,
        )

        self._set_running_state(
            snapshot.status == "running",
        )

    # Discrete observations
    # ==================================================================

    def _display_discrete(
        self,
        observations,
    ) -> None:
        self.discrete_table.setRowCount(
            0,
        )

            row = self.discrete_table.rowCount()

            )

                        observation.value,

                ),
            )

        self.discrete_table.resizeColumnsToContents()

    # Continuous observations
    # ==================================================================

    def _display_continuous(
        self,
        observations,
    ) -> None:
        current = self.feature_selector.currentText()

        self.feature_selector.blockSignals(
            True,
        )

        try:
            self.feature_selector.clear()

            names: list[str] = []

            for observation in observations:
                name = str(
                    observation.name,
                )

                if name and name not in names:
                    names.append(name)

            self.feature_selector.addItems(
                names,
            )

            if current in names:
                self.feature_selector.setCurrentText(
                    current,
                )

        finally:
            self.feature_selector.blockSignals(
                False,
            )

        self._render_selected_continuous(
            observations,
        )

    def _continuous_feature_changed(
        self,
        _name: str,
    ) -> None:
        snapshot = self._snapshot

        if snapshot is None:
            return

        self._render_selected_continuous(
            snapshot.continuous,
        )

    def _render_selected_continuous(
        self,
        observations,
    ) -> None:
        selected = self.feature_selector.currentText()

        if not selected:
            self.plot.clear()
            return

        values: list[float] = []

        for observation in observations:
            if observation.name != selected:
                continue

            value = observation.value

            if value is None:
                continue

            try:
                values.append(
                    float(value),
                )
            except (
                TypeError,
                ValueError,
            ):
                continue

        self.plot.set_data(
            values,
            selected,
        )

    # Events / traces
    # ==================================================================

    def _display_events(
        self,
        snapshot: ObservationSnapshot,
    ) -> None:
        self.event_table.setRowCount(
            0,
        )

        for event in snapshot.events:
            if not isinstance(event, TaskEvent):
                continue

            row = self.event_table.rowCount()

            self.event_table.insertRow(
                row,
            )

            self._set_event_row(
                row=row,
                event_type="event",
                name=event.name,
                source_or_path=event.source,
                payload_or_values=event.payload,
            )

        for trace in snapshot.traces:
            if not isinstance(trace, TracePoint):
                continue

            row = self.event_table.rowCount()

            self.event_table.insertRow(
                row,
            )

            values = {
                "discrete": dict(trace.discrete),
                "continuous": dict(trace.continuous),
            }

            self._set_event_row(
                row=row,
                event_type=trace.phase,
                name=trace.action,
                source_or_path=trace.path,
                payload_or_values=values,
            )

        self.event_table.resizeColumnsToContents()

    def _set_event_row(
        self,
        row: int,
        event_type: Any,
        name: Any,
        source_or_path: Any,
        payload_or_values: Any,
    ) -> None:
        values = self._safe_text(payload_or_values)

        self.event_table.setItem(
            row,
            0,
            QTableWidgetItem(
                self._safe_text(event_type),
            ),
        )

        self.event_table.setItem(
            row,
            1,
            QTableWidgetItem(
                self._safe_text(name),
            ),
        )

        self.event_table.setItem(
            row,
            2,
            QTableWidgetItem(
                self._safe_text(source_or_path),
            ),
        )

        self.event_table.setItem(
            row,
            3,
            QTableWidgetItem(
                values,
            ),
        )

    # ==================================================================
    # Replay / history
    # ==================================================================

    def refresh_history(self) -> None:
        try:
            replay = ReplayAPI(
                self.storage_root,
            )

            session_ids = replay.session_ids()

        except Exception as exc:
            self.status_label.setText(f"History error: {exc}")
            return

        self.history_list.clear()

        for session_id in session_ids:
            item = QListWidgetItem(
                str(session_id),
            )

            item.setData(
                Qt.UserRole,
                str(session_id),
            )

            self.history_list.addItem(
                item,
            )

    def _load_history_item(
        self,
        item: QListWidgetItem,
    ) -> None:
        session_id = item.data(
            Qt.UserRole,
        )

        if not session_id:
            return

        try:
            replay = ReplayAPI(
                self.storage_root,
            )

            snapshot = replay.session(
                str(session_id),
            )

            self._display_snapshot(
                snapshot,
            )

            self.status_label.setText(f"Loaded session: {session_id}")

            self._set_running_state(
                False,
            )

        except Exception as exc:
            self.status_label.setText(f"Replay error: {exc}")

    # ==================================================================
    # Helpers
    # ==================================================================

    @staticmethod
    def _safe_text(
        value: Any,
    ) -> str:
        if value is None:
            return "<missing>"

        try:
            return str(value)
        except Exception:
            return "<unprintable>"

    # ==================================================================
    # Lifecycle
    # ==================================================================

    def closeEvent(self, event) -> None:
        self.bridge.close()

        super().closeEvent(
            event,
        )

from threading import RLock

from .buckets import BucketManager

    EventTraceCollector,

class TaskObservationSession:
    """Live observation session owned by a separate observation thread/process."""

    def __init__(
        self,
        identity: TaskIdentity,
        store: SessionStore | None = None,
        buckets: BucketManager | None = None,
        feature_deriver: FeatureDeriver | None = None,
    ) -> None:
        self.identity = identity

        self.discrete = DiscreteFeatureCollector()
        self.continuous = ContinuousFeatureCollector()
        self.events = EventTraceCollector()

        self.store = store
        self.buckets = buckets
        self.feature_deriver = feature_deriver or FeatureDeriver()

        self._started_at = identity.started_at
        self._finished_at: str | None = None
        self._status = "running"
        self._metadata: dict[str, Any] = {}

        self._lock = RLock()

    def set_metadata(self, values: Mapping[str, Any]) -> None:
        with self._lock:
            self._metadata.update(values)

    def finish(
        self,
        status: str = "completed",
    ) -> ObservationSnapshot:
        with self._lock:
            self._finished_at = utc_now_iso()
            self._status = status

            # --------------------------------------------------
            # Build the complete raw observation snapshot first.
            #
            # All task events, traces, timestamps and raw
            # observations are now available to the derivation
            # layer.
            # --------------------------------------------------
            raw_snapshot = self.snapshot()

            # --------------------------------------------------
            # Derive higher-level observations.
            #
            # FeatureDeriver coordinates the independent
            # ToolObserver, NavigationObserver, TimingObserver,
            # and OutcomeObserver components.
            # --------------------------------------------------
            derived = self.feature_deriver.derive(raw_snapshot)

            # --------------------------------------------------
            # Persist derived continuous features.
            # --------------------------------------------------
            self.continuous.record_many(
                derived.continuous_features(),
                source="feature_deriver",
            )

            # --------------------------------------------------
            # Persist derived discrete features.
            # --------------------------------------------------
            self.discrete.record_many(
                derived.discrete_features(),
                source="feature_deriver",
            )

            # --------------------------------------------------
            # Rebuild the snapshot so the persisted observation
            # contains both raw and derived observations.
            # --------------------------------------------------
            snapshot = self.snapshot()

        if self.buckets is not None:
            self.buckets.add(snapshot)

        if self.store is not None:
            self.store.save(snapshot)

        return snapshot

    def snapshot(self) -> ObservationSnapshot:
        with self._lock:
            return ObservationSnapshot(
                identity=self.identity,
                discrete=self.discrete.snapshot(),
                continuous=self.continuous.snapshot(),
                events=self.events.events(),
                traces=self.events.traces(),
                started_at=self._started_at,
                finished_at=self._finished_at,
                status=self._status,
                metadata=self._metadata,
            )


class TaskObservationEngine:
    """Public facade used by filesystem, browser, and other adapters."""

    def __init__(
        self,
        storage_root: str | None = None,
        feature_deriver: FeatureDeriver | None = None,
    ) -> None:
        self.store = SessionStore(storage_root) if storage_root else None

        self.buckets = BucketManager()
        self.feature_deriver = feature_deriver or FeatureDeriver()

        self._sessions: dict[
            str,
            TaskObservationSession,
        ] = {}

        self._lock = RLock()

    def start(
        self,
        family_key: str,
        *,
        task_key: str = "",
        metadata: Mapping[str, Any] | None = None,
    ) -> TaskObservationSession:
        identity = TaskIdentity.create(
            family_key,
            task_key,
        )

        session = TaskObservationSession(
            identity,
            self.store,
            self.buckets,
            feature_deriver=self.feature_deriver,
        )

        if metadata:
            session.set_metadata(metadata)

        with self._lock:
            self._sessions[identity.session_id] = session

        return session

    def get(
        self,
        session_id: str,
    ) -> TaskObservationSession | None:
        with self._lock:
            return self._sessions.get(session_id)

    def finish(
        self,
        session_id: str,
        status: str = "completed",
    ) -> ObservationSnapshot | None:
        session = self.get(session_id)

        if session is None:
            return None

        snapshot = session.finish(status)

        with self._lock:
            self._sessions.pop(
                session_id,
                None,
            )

        return snapshot

    def replay(
        self,
        session_id: str,
    ) -> ObservationSnapshot:
        if self.store is None:
            raise RuntimeError("Persistent storage is not configured.")

        return self.store.replay(session_id)

from __future__ import annotations

from pathlib import Path
from typing import Iterable

from .models import ObservationSnapshot
from .session_store import SessionStore


class ReplayAPI:
    """Read-only API for visualization, experiments and mathematical adapters."""

    def __init__(self, root: str | Path) -> None:
        self.store = SessionStore(root)

    def session_ids(self) -> list[str]:
        return self.store.list_session_ids()

    def session(self, session_id: str) -> ObservationSnapshot:
        return self.store.load(session_id)

    def sessions(
        self, session_ids: Iterable[str] | None = None
    ) -> list[ObservationSnapshot]:
        ids = list(session_ids) if session_ids is not None else self.session_ids()
        return [self.session(session_id) for session_id in ids]

    def feature_rows(
        self, session_ids: Iterable[str] | None = None
    ) -> list[dict[str, object]]:
        rows: list[dict[str, object]] = []
        for snapshot in self.sessions(session_ids):
            row: dict[str, object] = {
                "session_id": snapshot.identity.session_id,
                "family_key": snapshot.identity.family_key,
                "status": snapshot.status,
            }
            for item in snapshot.discrete:
                row[f"discrete.{item.name}"] = item.value
            for item in snapshot.continuous:
                if item.value is not None:
                    row[f"continuous.{item.name}"] = item.value
            rows.append(row)
        return rows

from __future__ import annotations

from threading import Lock
from typing import Any, Mapping
from datetime import datetime, timezone

from .models import ContinuousObservation, DiscreteObservation, TaskEvent, TracePoint


class DiscreteFeatureCollector:
    """Thread-safe collector for categorical/state-like task features."""

    def __init__(self) -> None:
        self._items: list[DiscreteObservation] = []
        self._lock = Lock()

    def record(self, name: str, value: str | int | bool | None, *, source: str = "unknown", step: int | None = None) -> None:
        with self._lock:
            self._items.append(DiscreteObservation(name, value, source=source, step=step))

    def record_many(self, values: Mapping[str, str | int | bool | None], *, source: str = "unknown", step: int | None = None) -> None:
        with self._lock:
            for name, value in values.items():
                self._items.append(DiscreteObservation(name, value, source=source, step=step))

    def snapshot(self) -> tuple[DiscreteObservation, ...]:
        with self._lock:
            return tuple(self._items)

    def clear(self) -> None:
        with self._lock:
            self._items.clear()


class ContinuousFeatureCollector:
    """Thread-safe collector for numeric task features.

    Missing/invalid numeric values are accepted and represented as None rather
    than terminating the observation process.
    """

    def __init__(self) -> None:
        self._items: list[ContinuousObservation] = []
        self._lock = Lock()

    def record(self, name: str, value: float | int | None, *, source: str = "unknown", step: int | None = None) -> None:
        safe_value: float | None
        try:
            safe_value = None if value is None else float(value)
        except (TypeError, ValueError):
            safe_value = None
        with self._lock:
            self._items.append(ContinuousObservation(name, safe_value, source=source, step=step))

    def record_many(self, values: Mapping[str, float | int | None], *, source: str = "unknown", step: int | None = None) -> None:
        for name, value in values.items():
            self.record(name, value, source=source, step=step)

    def snapshot(self) -> tuple[ContinuousObservation, ...]:
        with self._lock:
            return tuple(self._items)

    def clear(self) -> None:
        with self._lock:
            self._items.clear()


class EventTraceCollector:
    """Thread-safe event and trace recorder."""

    def __init__(self) -> None:
        self._events: list[TaskEvent] = []
        self._traces: list[TracePoint] = []
        self._lock = Lock()

    def event(self, name: str, *, source: str = "unknown", payload: Mapping[str, Any] | None = None, step: int | None = None) -> None:
        with self._lock:
            self._events.append(TaskEvent(name=name, source=source, payload=payload or {}, step=step))

    def trace(
        self,
        phase: str,
        *,
        action: str = "",
        path: str | None = None,
        discrete: Mapping[str, str | int | bool | None] | None = None,
        continuous: Mapping[str, float | None] | None = None,
    ) -> int:
        with self._lock:
            sequence = len(self._traces)
            self._traces.append(
                TracePoint(
                    sequence=sequence,
                    timestamp=datetime.now(timezone.utc).isoformat(),
                    phase=phase,
                    action=action,
                    path=path,
                    discrete=discrete or {},
                    continuous=continuous or {},
                )
            )
            return sequence

    def events(self) -> tuple[TaskEvent, ...]:
        with self._lock:
            return tuple(self._events)

    def traces(self) -> tuple[TracePoint, ...]:
        with self._lock:
            return tuple(self._traces)

from __future__ import annotations

from collections import defaultdict
from threading import Lock
from typing import Iterable

from .models import BucketKey, ObservationBucket, ObservationSnapshot


class BucketManager:
    """Groups sessions by stable categorical features without altering raw data."""

    def __init__(self) -> None:
        self._buckets: dict[BucketKey, ObservationBucket] = {}
        self._lock = Lock()

    def add(self, snapshot: ObservationSnapshot) -> None:
        session_id = snapshot.identity.session_id
        values: dict[str, set[str]] = defaultdict(set)
        for item in snapshot.discrete:
            values[item.name].add(self._normalize(item.value))

        with self._lock:
            for name, bucket_values in values.items():
                for value in bucket_values:
                    key = BucketKey(name, value)
                    bucket = self._buckets.setdefault(key, ObservationBucket(key))
                    bucket.add(session_id)

    def get(self, name: str, value: object) -> ObservationBucket | None:
        with self._lock:
            bucket = self._buckets.get(BucketKey(name, self._normalize(value)))
            if bucket is None:
                return None
            return ObservationBucket(bucket.key, bucket.count, list(bucket.sessions))

    def all(self) -> tuple[ObservationBucket, ...]:
        with self._lock:
            return tuple(ObservationBucket(b.key, b.count, list(b.sessions)) for b in self._buckets.values())

    @staticmethod
    def _normalize(value: object) -> str:
        return "<MISSING>" if value is None else str(value)

    if app is None:
        app = QCoreApplication([])

    return app


def wait_for_spy(
    app,
    spy: QSignalSpy,
    timeout_ms: int = 3000,
) -> bool:
    if spy.count() > 0:
        return True

    deadline = time.monotonic() + (timeout_ms / 1000.0)

    while time.monotonic() < deadline:
        app.processEvents()

        if spy.count() > 0:
            return True

        time.sleep(0.005)

    return False


def test_bridge_forwards_start_event_and_finish(
    tmp_path,
    qt_core_app,
) -> None:
    bridge = ObservationBridge(
        str(tmp_path),
    )

    started = QSignalSpy(
        bridge.task_started,
    )

    updated = QSignalSpy(
        bridge.observation_updated,
    )

    finished = QSignalSpy(
        bridge.task_finished,
    )

    errors = QSignalSpy(
        bridge.error,
    )

    try:
        bridge.start_task(
            "filesystem",
            metadata={
                "agent_type": "filesystem",
            },
        )

        assert wait_for_spy(
            qt_core_app,
            started,
        )

        assert started.at(0)[0] == "filesystem"

        bridge.record_event(
            "tool_started",
            {
                "tool_name": "list_directory",
                "path": ".",
            },
        )

        bridge.record_event(
            "tool_completed",
            {
                "tool_name": "list_directory",
                "path": ".",
                "success": True,
            },
        )

        bridge.finish_task()

        assert wait_for_spy(
            qt_core_app,
            finished,
        )

        assert errors.count() == 0

        snapshot = finished.at(0)[0]

        assert isinstance(
            snapshot,
            ObservationSnapshot,

        assert snapshot.status == "completed"

        event_names = [event.name for event in snapshot.events]

        assert "tool_started" in event_names
        assert "tool_completed" in event_names

        continuous = {item.name: item.value for item in snapshot.continuous}

        assert continuous["total_tool_calls"] == 1.0
        assert continuous["successful_tool_calls"] == 1.0

        assert updated.count() >= 2

    finally:
        bridge.close()

        item
        for item in snapshot.continuous
        if item.name == "bad_number" and item.source == "unknown"
    ]

    assert len(bad_number) == 1

    # ----------------------------------------------------------

    # =============================================================

    def handle_new_task(
        self,
        intent: str,
        task_prompt: str,
    ):
        """
        Route a classified task to the appropriate agent.

        MainWindow coordinates the UI and lifecycle routing.
        Authorization itself belongs to ApprovalEngine.
        """

        # ---------------------------------------------------------
        # 1. Prevent concurrent tasks
        # ---------------------------------------------------------

        if self.agent_worker is not None and self.agent_worker.isRunning():
            error_html = ChatMessageFormatter.format_message(
                "error",
                "A task is already running. " "Cancel it before starting another task.",
            )

            self.chat_window.chat_display.append(error_html)

            return

        if self.pending_approval_task is not None:
            error_html = ChatMessageFormatter.format_message(
                "error",
                "A task is waiting for human approval. "
                "Approve or deny it before starting another task.",
            )

            self.chat_window.chat_display.append(error_html)

            return

        # ---------------------------------------------------------
        # 2. Create canonical Task
        # ---------------------------------------------------------

        task = Task(prompt=task_prompt)

        self.current_task = task

        self.task_bar.task_list.addItem(f"📌 {task_prompt}")

        # ---------------------------------------------------------
        # 3. Approval layer availability check
        # ---------------------------------------------------------

        if self.approval_engine is None:
            error_html = ChatMessageFormatter.format_message(
                "error",
                "Approval engine is unavailable because "
                "GEMINI_API_KEY is not configured.",
            )

            self.chat_window.chat_display.append(error_html)

            return

        # ---------------------------------------------------------
        # 4. Inform UI that approval evaluation has started
        # ---------------------------------------------------------

        approval_started_html = ChatMessageFormatter.format_message(
            "system",
            "🔐 Evaluating task authorization...",
        )

        self.chat_window.chat_display.append(approval_started_html)

        # ---------------------------------------------------------
        # 5. Run approval classification + policy evaluation
        # ---------------------------------------------------------

        try:
            decision = self.approval_engine.evaluate(task_prompt)

        except Exception as exc:
            error_html = ChatMessageFormatter.format_message(
                "error",
                "Approval classification failed: " f"{type(exc).__name__}: {exc}",
            )

            self.chat_window.chat_display.append(error_html)

            return

        # ---------------------------------------------------------
        # 6. Display classification
        # ---------------------------------------------------------

        classification = decision.classification

        classification_msg = (
            "🔍 Approval classification:\n"
            f"- Operation: `{classification.operation_type}`\n"
            f"- Data sensitivity: "
            f"`{classification.data_sensitivity}`\n"
            f"- External side effect: "
            f"`{classification.external_side_effect}`\n"
            f"- Destructive: "
            f"`{classification.destructive}`\n"
            f"- Credential use: "
            f"`{classification.credential_use}`"
        )

        self.chat_window.chat_display.append(
            ChatMessageFormatter.format_message(
                "system",
                classification_msg,
            )
        )

        # ---------------------------------------------------------
        # 7. Prepare policy reasons
        # ---------------------------------------------------------

        reasons = "\n".join(f"- {reason}" for reason in decision.reasons)

        # ---------------------------------------------------------
        # 8. DENY
        # ---------------------------------------------------------

        if decision.status == ApprovalStatus.DENY:
            self.chat_window.chat_display.append(
                ChatMessageFormatter.format_message(
                    "error",
                    "🛑 Task denied by the local approval policy.\n" f"{reasons}",
                )
            )

            return
        # ---------------------------------------------------------
        # 9. REVIEW
        # ---------------------------------------------------------

        if decision.status == ApprovalStatus.REVIEW:

            # Keep the canonical Task object alive while the
            # human approval UI is displayed.
            self.pending_approval_task = (
                task,
                intent,
            )

            self.chat_window.chat_display.append(
                ChatMessageFormatter.format_message(
                    "system",
                    "⏸️ Human approval required before "
                    "browser execution.\n"
                    f"{reasons}",
                )
            )

            # Display the actual approval controls.
            #
            # The ChatWindow owns the presentation.
            # MainWindow remains responsible for deciding what
            # happens after Approve / Deny.
            self.chat_window.show_approval_request(decision)

            return

        # ---------------------------------------------------------
        # 10. APPROVED WITHOUT HUMAN REVIEW
        # ---------------------------------------------------------

        self.chat_window.chat_display.append(
            ChatMessageFormatter.format_message(
                "system",
                (
                    f"✅ Task approved for {intent.replace('_', ' ').title()}.\n"
                    f"{reasons}"
                ),
            )
        )

        self.dispatch_task(task, intent)
        # =============================================================

    def dispatch_task(
        self,
        task: Task,
        intent: str,
    ):
        """
        Route an approved Task to the correct execution worker.

        MainWindow does not implement agent behavior.
        It only selects the appropriate agent boundary.
        """

        if task is None:
            self.chat_window.chat_display.append(
                ChatMessageFormatter.format_message(
                    "error",
                    "Cannot dispatch task: task is missing.",
                )
            )
            return

        if intent == "BROWSER_TASK":
            self.start_browser_task(task)
            return

        if intent == "FILESYSTEM_TASK":
            self.start_filesystem_task(task)
            return

        self.chat_window.chat_display.append(
            ChatMessageFormatter.format_message(
                "error",
                f"Unsupported task intent: {intent}",
            )
        )

    # =============================================================
    # Browser task execution
    # =============================================================

    def start_browser_task(self, task: Task):
        """
        Start the browser agent for an approved Task.

        This method is intentionally separate from the approval
        system.

        It can be called by:

        1. The automatic ALLOW path.
        2. The human APPROVE path.

        Approval decisions are made before this method is called.
        """

        if task is None:
            self.chat_window.chat_display.append(
                ChatMessageFormatter.format_message(
                    "error",
                    "Cannot start browser task: task is missing.",
                )
            )

            return

        # ---------------------------------------------------------
        # Prevent concurrent browser execution
        # ---------------------------------------------------------

        if self.agent_worker is not None and self.agent_worker.isRunning():
            self.chat_window.chat_display.append(
                ChatMessageFormatter.format_message(
                    "error",
                    "A browser task is already running.",
                )
            )

            return

        # ---------------------------------------------------------
        # Inform UI
        # ---------------------------------------------------------

        sys_msg = ChatMessageFormatter.format_message(
            "system",
            "Connecting browser agent to embedded Qt WebEngine "
            f"over CDP (`http://127.0.0.1:{CDP_PORT}`)...",
        )

        self.chat_window.chat_display.append(sys_msg)

        # ---------------------------------------------------------
        # Gemini API key
        # ---------------------------------------------------------

        api_key = os.getenv("GEMINI_API_KEY")

        if not api_key:
            err_msg = ChatMessageFormatter.format_message(
                "error",
                "GEMINI_API_KEY environment variable is missing.",
            )

            self.chat_window.chat_display.append(err_msg)

            return

        # ---------------------------------------------------------
        # Connect browser-use to embedded Qt WebEngine
        # ---------------------------------------------------------

        cdp_url = f"http://127.0.0.1:{CDP_PORT}"

        # ---------------------------------------------------------
        # Start browser agent worker
        # ---------------------------------------------------------

        self.agent_worker = AgentExecutionWorker(
            task,
            api_key,
            cdp_url=cdp_url,
        )
        self.active_agent_type = "BROWSER_AGENT"

        self.agent_worker.step_updated.connect(self.on_agent_step)

        self.agent_worker.task_finished.connect(self.on_agent_finished)

        self.agent_worker.start()

            )

            return

        self.pending_approval_task = None

        task, intent = pending

        self.chat_window.chat_display.append(
            ChatMessageFormatter.format_message(
                "system",
                "✅ Human approval granted. "
                f"Starting {intent.replace('_', ' ').title()}...",
            )
        )

        self.dispatch_task(
            task,
            intent,
        )

    def on_approval_denied(self):
        """
        Stop the pending task after the human presses Deny.
        """

        self.pending_approval_task = None

        self.current_task = None

        self.chat_window.chat_display.append(
            ChatMessageFormatter.format_message(
                "system",
                "🛑 Human approval denied. " "The task will not be executed.",
            )
        )

    # =============================================================
    # Agent step updates
    # =============================================================

    def on_agent_step(self, step_msg: str):
        """
        Display a step update from the active agent.

        Agent workers are responsible for identifying themselves.
        MainWindow only presents the emitted message.
        """

        agent_html = ChatMessageFormatter.format_message(
            "agent",
            step_msg,
        )

        self.chat_window.chat_display.append(agent_html)

    # =============================================================
    # Agent completion
    # =============================================================

    def on_agent_finished(self, result):
        """
        Display the final browser-agent execution result.
        """

        # Generate formatted HTML block from response parser.
        parsed_content = AgentResponseParser.parse_and_format(result)

        # Wrap system header card around parsed content
        # without escaping the inner HTML.
        header_html = ChatMessageFormatter.format_message(
            "system",
            "**Task execution finished.**",
        )

        self.chat_window.chat_display.append(header_html)

        self.chat_window.chat_display.append(parsed_content)
        self.active_agent_type = None

    # =============================================================
    # Filesystem completion
    # =============================================================

    def on_filesystem_finished(self, result):
        """
        Display the final filesystem-agent response.
        """

        final_text = ""

        if isinstance(result, dict):
            final_text = result.get(
                "final_text",
                "",
            )

        if not final_text:
            final_text = "Filesystem task completed successfully."

        self.chat_window.chat_display.append(
            ChatMessageFormatter.format_message(
                "system",
                "**Filesystem task finished.**",
            )
        )

        self.chat_window.chat_display.append(
            ChatMessageFormatter.format_message(
                "agent",
                final_text,
            )
        )
        self.active_agent_type = None

    # =============================================================
    # Filesystem errors
    # =============================================================

    def on_filesystem_error(self, error_message: str):
        """
        Display filesystem-agent execution errors.
        """

        self.chat_window.chat_display.append(
            ChatMessageFormatter.format_message(
                "error",
                f"Filesystem Agent Error: {error_message}",
            )
        )
        self.active_agent_type = None

    # =============================================================
    # Task cancellation
    # =============================================================

    def cancel_current_task(self):
        """
        Request cancellation of the currently running task.
        """

        if self.agent_worker is None:
            return

        self.agent_worker.cancel()

# app_menu_bar.py
from PySide6.QtWidgets import QMenuBar, QMessageBox, QWidget


class AppMenuBar(QMenuBar):
    """Custom Application Menu Bar providing agent diagnostics and status monitoring."""

    def __init__(self, parent: QWidget = None):
        super().__init__(parent)
        self.main_window = parent
        self._init_menus()

    def _init_menus(self):
        # 1. Agent / Status Menu
        agent_menu = self.addMenu("Agent")

        # Action: Check Agent Diagnostics
        status_action = agent_menu.addAction("🔍 Agent Diagnostics & Status")
        status_action.setShortcut("Ctrl+D")
        status_action.triggered.connect(self.show_agent_status_dialog)
        # 2. Observation Menu
        observation_menu = self.addMenu("Observation")

        # Action: Open Observation Explorer
        observation_action = observation_menu.addAction("📊 Observation Explorer")
        observation_action.setShortcut("Ctrl+Shift+O")
        observation_action.triggered.connect(self.show_observation_widget)

    def show_agent_status_dialog(self):
        """Fetches status from main window and displays a diagnostic popup."""
        status_info = {
            "status": "idle",
            "current_step": 0,
            "last_action": "No agent running",
            "error": None,
        }

        # Query main window if available
        if (
            self.main_window
            and hasattr(self.main_window, "get_agent_status")
            and callable(self.main_window.get_agent_status)
        ):
            status_info = self.main_window.get_agent_status()

        # Format diagnostic message
        status_str = status_info.get("status", "unknown").upper()
        step_count = status_info.get("current_step", 0)
        last_act = status_info.get("last_action", "N/A")
        err = status_info.get("error", None)

        msg = (
            f"<b>Agent Status:</b> {status_str}<br>"
            f"<b>Steps Executed:</b> {step_count}<br>"
            f"<b>Last Action:</b> {last_act}"
        )

        if err:
            msg += f"<br><br><span style='color: red;'><b>Error:</b> {err}</span>"

        # Display popup dialog
        QMessageBox.information(self.main_window, "Agent Diagnostics", msg)
        # Display popup dialog
        QMessageBox.information(self.main_window, "Agent Diagnostics", msg)

    def show_observation_widget(self):
        """Open the task observation visualization window."""
        if (
            self.main_window
            and hasattr(self.main_window, "show_observation_widget")
            and callable(self.main_window.show_observation_widget)
        ):
            self.main_window.show_observation_widget()
