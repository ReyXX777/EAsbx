from __future__ import annotations

from typing import Any, Mapping

from PySide6.QtCore import QObject, QThread, Signal, Slot

from .observation_thread import ObservationWorker


class ObservationBridge(QObject):
    """Thread-safe Qt bridge for the observation worker.

    UI code communicates only with this object.

    The bridge owns the worker thread and forwards requests to the
    ObservationWorker using Qt's queued signal/slot mechanism.
    """

    task_started = Signal(str)
    observation_updated = Signal(object)
    task_finished = Signal(object)
    error = Signal(str)

    request_start = Signal(str, str, object)
    request_discrete = Signal(str, object)
    request_continuous = Signal(str, object)
    request_event = Signal(str, object)
    request_snapshot = Signal()
    request_finish = Signal()
    request_shutdown = Signal()

    def __init__(
        self,
        storage_root: str,
        parent: QObject | None = None,
    ) -> None:
        super().__init__(parent)

        self._closed = False

        self._thread = QThread(self)

        self._worker = ObservationWorker(
            storage_root,
        )

        self._worker.moveToThread(
            self._thread,
        )

        # --------------------------------------------------------------
        # Requests: UI -> worker
        # --------------------------------------------------------------

        self.request_start.connect(
            self._worker.start_task,
        )

        self.request_discrete.connect(
            self._worker.record_discrete,
        )

        self.request_continuous.connect(
            self._worker.record_continuous,
        )

        self.request_event.connect(
            self._worker.record_event,
        )

        self.request_snapshot.connect(
            self._worker.publish_current_state,
        )

        self.request_finish.connect(
            self._worker.finish_task,
        )

        self.request_shutdown.connect(
            self._worker.shutdown,
        )

        # --------------------------------------------------------------
        # Results: worker -> UI
        # --------------------------------------------------------------

        self._worker.task_started.connect(
            self.task_started,
        )

        self._worker.observation_updated.connect(
            self.observation_updated,
        )

        self._worker.task_finished.connect(
            self.task_finished,
        )

        self._worker.worker_error.connect(
            self.error,
        )

        self._thread.start()

    # ------------------------------------------------------------------
    # Public commands
    # ------------------------------------------------------------------

    @Slot(str, str, object)
    def start_task(
        self,
        family_key: str,
        task_key: str = "",
        metadata: Mapping[str, Any] | None = None,
    ) -> None:
        """Request creation of a new observation session."""

        if self._closed:
            return

        self.request_start.emit(
            family_key,
            task_key,
            metadata,
        )

    @Slot(str, object)
    def record_discrete(
        self,
        name: str,
        value: Any = None,
    ) -> None:
        """Request a discrete observation."""

        if self._closed:
            return

        self.request_discrete.emit(
            name,
            value,
        )

    @Slot(str, object)
    def record_continuous(
        self,
        name: str,
        value: Any = None,
    ) -> None:
        """Request a continuous observation."""

        if self._closed:
            return

        self.request_continuous.emit(
            name,
            value,
        )

    @Slot(str, object)
    def record_event(
        self,
        event_type: str,
        payload: Any = None,
    ) -> None:
        """Request an event observation."""

        if self._closed:
            return

        self.request_event.emit(
            event_type,
            payload,
        )

    def request_current_snapshot(self) -> None:
        """Request publication of the current live snapshot."""

        if self._closed:
            return

        self.request_snapshot.emit()

    def finish_task(self) -> None:
        """Request finalization of the active observation session."""

        if self._closed:
            return

        self.request_finish.emit()

    # ------------------------------------------------------------------
    # Lifecycle
    # ------------------------------------------------------------------

    def close(self) -> None:
        """Stop the worker thread safely and release Qt resources."""

        if self._closed:
            return

        self._closed = True

        if not self._thread.isRunning():
            return

        # The shutdown request is queued to the worker so that the
        # session reference is released on the worker thread.
        self.request_shutdown.emit()

        # Stop the event loop after queued work has been processed.
        self._thread.quit()

        if not self._thread.wait(3000):
            self._thread.terminate()
            self._thread.wait(1000)

    def __del__(self) -> None:
        try:
            self.close()
        except Exception:
            pass

from __future__ import annotations

from typing import Any, Mapping

from PySide6.QtCore import QObject, Signal, Slot

from ..session import TaskObservationEngine, TaskObservationSession


class ObservationWorker(QObject):
    """Owns live observation state on the dedicated observation thread.

    The worker is the only Qt-side object allowed to mutate the active
    TaskObservationSession.

    The presentation layer communicates with this object through queued
    signals exposed by ObservationBridge.

    Responsibilities:

    - create live observation sessions
    - mutate collectors belonging to the active session
    - publish current snapshots
    - finish sessions
    - release the active session

    Persistence is handled by TaskObservationSession through the current
    SessionStore/storage architecture.
    """

    task_started = Signal(str)
    observation_updated = Signal(object)
    task_finished = Signal(object)
    worker_error = Signal(str)

    def __init__(
        self,
        storage_root: str,
        parent: QObject | None = None,
    ) -> None:
        super().__init__(parent)

        self._engine = TaskObservationEngine(
            storage_root=storage_root,
        )

        self._session: TaskObservationSession | None = None

    # ------------------------------------------------------------------
    # Session lifecycle
    # ------------------------------------------------------------------

    @Slot(str, str, object)
    def start_task(
        self,
        family_key: str,
        task_key: str = "",
        metadata: Mapping[str, Any] | None = None,
    ) -> None:
        """Start a new observation session."""

        try:
            if self._session is not None:
                self.worker_error.emit(
                    "Cannot start a new task while another task is active."
                )
                return

            family_key = (family_key or "").strip()

            if not family_key:
                self.worker_error.emit(
                    "Cannot start an observation without a task family."
                )
                return

            task_key = (task_key or "").strip()

            self._session = self._engine.start(
                family_key,
                task_key=task_key,
                metadata=metadata,
            )

            self.task_started.emit(
                family_key,
            )

            self.observation_updated.emit(
                self._session.snapshot(),
            )

        except Exception as exc:
            self.worker_error.emit(f"Task start error: {type(exc).__name__}: {exc}")

    @Slot()
    def finish_task(self) -> None:
        """Finish the active observation session."""

        session = self._session

        if session is None:
            self.worker_error.emit("No active observation session.")
            return

        try:
            snapshot = session.finish()

            self._session = None

            self.task_finished.emit(
                snapshot,
            )

            self.observation_updated.emit(
                snapshot,
            )

        except Exception as exc:
            self.worker_error.emit(
                f"Task finalization error: {type(exc).__name__}: {exc}"
            )

    # ------------------------------------------------------------------
    # Discrete observations
    # ------------------------------------------------------------------

    @Slot(str, object)
    def record_discrete(
        self,
        name: str,
        value: Any = None,
    ) -> None:
        """Record a discrete observation on the active session."""

        session = self._session

        if session is None:
            self.worker_error.emit("No active observation session.")
            return

        try:
            session.discrete.record(
                name,
                value,
            )

        except Exception as exc:
            self.worker_error.emit(
                "Discrete observation error: " f"{type(exc).__name__}: {exc}"
            )

    # ------------------------------------------------------------------
    # Continuous observations
    # ------------------------------------------------------------------

    @Slot(str, object)
    def record_continuous(
        self,
        name: str,
        value: Any = None,
    ) -> None:
        """Record a continuous observation on the active session."""

        session = self._session

        if session is None:
            self.worker_error.emit("No active observation session.")
            return

        try:
            session.continuous.record(
                name,
                value,
            )

        except Exception as exc:
            self.worker_error.emit(
                "Continuous observation error: " f"{type(exc).__name__}: {exc}"
            )

    # ------------------------------------------------------------------
    # Events
    # ------------------------------------------------------------------

    @Slot(str, object)
    def record_event(
        self,
        event_type: str,
        payload: Any = None,
    ) -> None:
        """Record an event through the current event collector."""

        session = self._session

        if session is None:
            self.worker_error.emit("No active observation session.")
            return

        try:
            session.events.record(
                event_type,
                payload,
            )

        except Exception as exc:
            self.worker_error.emit(
                "Event observation error: " f"{type(exc).__name__}: {exc}"
            )

    # ------------------------------------------------------------------
    # Snapshot publication
    # ------------------------------------------------------------------

    @Slot()
    def publish_current_state(self) -> None:
        """Publish the current raw observation snapshot."""

        session = self._session

        if session is None:
            return

        try:
            snapshot = session.snapshot()

            self.observation_updated.emit(
                snapshot,
            )

        except Exception as exc:
            self.worker_error.emit(f"Snapshot error: {type(exc).__name__}: {exc}")

    # ------------------------------------------------------------------
    # Shutdown
    # ------------------------------------------------------------------

    @Slot()
    def shutdown(self) -> None:
        """Release the active session reference.

        A live unfinished session is deliberately not finalized here.
        Normal persistence occurs through finish_task().
        """

        self._session = None

from __future__ import annotations

import math
from typing import Iterable

from PySide6.QtCore import QPointF, Qt
from PySide6.QtGui import QPainter
from PySide6.QtWidgets import QWidget


class ObservationPlot(QWidget):
    """Small dependency-free Qt plot for continuous observations.

    The plot is intentionally independent of:

    - TaskObservationSession
    - TaskObservationEngine
    - SessionStore
    - ReplayAPI
    - collector implementations

    It only receives numeric values and renders them.
    """

    def __init__(
        self,
        parent: QWidget | None = None,
    ) -> None:
        super().__init__(parent)

        self.setMinimumHeight(220)

        self._values: list[float] = []
        self._title = "Continuous Feature"

    def set_data(
        self,
        values: Iterable[float],
        title: str = "Continuous Feature",
    ) -> None:
        """Replace the plotted values."""

        cleaned: list[float] = []

        for value in values:
            try:
                numeric = float(value)
            except (TypeError, ValueError):
                continue

            if not math.isfinite(numeric):
                continue

            cleaned.append(numeric)

        self._values = cleaned
        self._title = str(title)

        self.update()

    def clear(self) -> None:
        """Clear the current plot."""

        self._values.clear()
        self.update()

    def paintEvent(self, _event) -> None:
        painter = QPainter(self)

        try:
            painter.setRenderHint(
                QPainter.Antialiasing,
                True,
            )

            rect = self.rect().adjusted(
                45,
                25,
                -20,
                -35,
            )

            painter.drawText(
                10,
                18,
                self._title,
            )

            if not self._values:
                painter.drawText(
                    rect,
                    Qt.AlignCenter,
                    "No numeric observations.",
                )
                return

            minimum = min(self._values)
            maximum = max(self._values)

            if math.isclose(
                minimum,
                maximum,
            ):
                padding = max(
                    abs(minimum) * 0.05,
                    1.0,
                )

                minimum -= padding
                maximum += padding

            painter.drawRect(
                rect,
            )

            point_count = len(self._values)

            points: list[QPointF] = []

            for index, value in enumerate(self._values):
                if point_count == 1:
                    x = float(rect.center().x())
                else:
                    x = rect.left() + (index / (point_count - 1)) * rect.width()

                normalized = (value - minimum) / (maximum - minimum)

                y = rect.bottom() - normalized * rect.height()

                points.append(
                    QPointF(
                        x,
                        y,
                    )
                )

            if len(points) >= 2:
                painter.drawPolyline(
                    points,
                )
            else:
                painter.drawEllipse(
                    points[0],
                    3,
                    3,
                )

            painter.drawText(
                5,
                rect.top(),
                f"{maximum:.4g}",
            )

            painter.drawText(
                5,
                rect.bottom(),
                f"{minimum:.4g}",
            )

            painter.drawText(
                rect.left(),
                rect.bottom() + 25,
                "start",
            )

            painter.drawText(
                rect.right() - 35,
                rect.bottom() + 25,
                "end",
            )

        finally:
            painter.end()

        )

            self._on_task_finished,
        )

        self.bridge.error.connect(
            self._on_error,
        )

    # ==================================================================
    # Commands
    # ==================================================================

    def _start_task(self) -> None:
        family = self.family_field.currentText().strip()

        if not family:
            self.status_label.setText("Enter a task family.")
            return

        self.bridge.start_task(
            family,
        )

    def _finish_task(self) -> None:
        self.bridge.finish_task()

    # ==================================================================
    # Worker events
    # ==================================================================

    def _on_task_started(
        self,
        family: str,
    ) -> None:
        self.status_label.setText(f"Collecting: {family}")

        self.family_label.setText(
            family,
        )

        self.session_status_label.setText(
            "running",
        )

        self._set_running_state(
            True,
        )

    def _on_task_finished(
        self,
        snapshot: object,
    ) -> None:
        if not isinstance(
            snapshot,
            ObservationSnapshot,
        ):
            self._on_error("Worker returned an invalid observation snapshot.")
            return

        self.status_label.setText("Observation completed.")

        self._display_snapshot(
            snapshot,
        )

        self._set_running_state(
            False,
        )

        self.refresh_history()

    def _on_error(
        self,
        message: str,
    ) -> None:
        self.status_label.setText(f"Observation error: {message}")

    def _set_running_state(
        self,
        running: bool,
    ) -> None:
        self.start_button.setEnabled(
            not running,
        )

        self.finish_button.setEnabled(
            running,
        )

    # ==================================================================
    # Snapshot display
    # ==================================================================

    def _display_snapshot(
        self,
        snapshot: object,
    ) -> None:
        if not isinstance(
            snapshot,
            ObservationSnapshot,
        ):
            self._on_error("Received an invalid observation snapshot.")
            return

        self._snapshot = snapshot

        identity = snapshot.identity

        self.session_id_label.setText(
            str(identity.session_id),
        )

        self.family_label.setText(
            str(identity.family_key),
        )

        self.session_status_label.setText(
            str(snapshot.status),
        )

        self.started_at_label.setText(
            self._safe_text(snapshot.started_at),
        )

        self.finished_at_label.setText(
            self._safe_text(snapshot.finished_at),
        )

        self._display_discrete(
            snapshot.discrete,
        )

        self._display_continuous(
            snapshot.continuous,
        )

        self._display_events(
            snapshot,
        )

        if snapshot.status == "running":
            self._set_running_state(
                True,
            )
        else:
            self._set_running_state(
                False,
            )

    # ==================================================================
    # Discrete observations
    # ==================================================================

    def _display_discrete(
        self,
        observations,
    ) -> None:
        self.discrete_table.setRowCount(
            0,
        )

        for observation in observations:
            row = self.discrete_table.rowCount()

            self.discrete_table.insertRow(
                row,
            )

            self.discrete_table.setItem(
                row,
                0,
                QTableWidgetItem(
                    self._safe_text(
                        observation.name,
                    )
                ),
            )

            self.discrete_table.setItem(
                row,
                1,
                QTableWidgetItem(
                    self._safe_text(
                        observation.value,
                    )
                ),
            )

        self.discrete_table.resizeColumnsToContents()

    # ==================================================================
    # Continuous observations
    # ==================================================================

    def _display_continuous(
        self,
        observations,
    ) -> None:
        current = self.feature_selector.currentText()

        self.feature_selector.blockSignals(
            True,
        )

        try:
            self.feature_selector.clear()

            names: list[str] = []

            for observation in observations:
                name = str(
                    observation.name,
                )

                if name and name not in names:
                    names.append(name)

            self.feature_selector.addItems(
                names,
            )

            if current in names:
                self.feature_selector.setCurrentText(
                    current,
                )

        finally:
            self.feature_selector.blockSignals(
                False,
            )

        self._render_selected_continuous(
            observations,
        )

    def _continuous_feature_changed(
        self,
        _name: str,
    ) -> None:
        snapshot = self._snapshot

        if snapshot is None:
            return

        self._render_selected_continuous(
            snapshot.continuous,
        )

    def _render_selected_continuous(
        self,
        observations,
    ) -> None:
        selected = self.feature_selector.currentText()

        if not selected:
            self.plot.clear()
            return

        values: list[float] = []

        for observation in observations:
            if observation.name != selected:
                continue

            value = observation.value

            if value is None:
                continue

            try:
                values.append(
                    float(value),
                )
            except (
                TypeError,
                ValueError,
            ):
                continue

        self.plot.set_data(
            values,
            selected,
        )

    # ==================================================================
    # Events / traces
    # ==================================================================

    def _display_events(
        self,
        snapshot: ObservationSnapshot,
    ) -> None:
        self.event_table.setRowCount(
            0,
        )

        # Current ObservationSnapshot explicitly exposes both event
        # collections. They are rendered in one presentation table.
        collections = (
            snapshot.events,
            snapshot.traces,
        )

        for collection in collections:
            for event in collection:
                row = self.event_table.rowCount()

                self.event_table.insertRow(
                    row,
                )

                event_type = getattr(
                    event,
                    "event_type",
                    getattr(
                        event,
                        "type",
                        "",
                    ),
                )

                name = getattr(
                    event,
                    "name",
                    getattr(
                        event,
                        "operation",
                        "",
                    ),
                )

                value = getattr(
                    event,
                    "value",
                    getattr(
                        event,
                        "payload",
                        "",
                    ),
                )

                self.event_table.setItem(
                    row,
                    0,
                    QTableWidgetItem(
                        self._safe_text(
                            event_type,
                        )
                    ),
                )

                self.event_table.setItem(
                    row,
                    1,
                    QTableWidgetItem(
                        self._safe_text(
                            name,
                        )
                    ),
                )

                self.event_table.setItem(
                    row,
                    2,
                    QTableWidgetItem(
                        self._safe_text(
                            value,
                        )
                    ),
                )

        self.event_table.resizeColumnsToContents()

    # ==================================================================
    # Replay / history
    # ==================================================================

    def refresh_history(self) -> None:
        try:
            replay = ReplayAPI(
                self.storage_root,
            )

            session_ids = replay.session_ids()

        except Exception as exc:
            self.status_label.setText(f"History error: {exc}")
            return

        self.history_list.clear()

        for session_id in session_ids:
            item = QListWidgetItem(
                str(session_id),
            )

            item.setData(
                Qt.UserRole,
                str(session_id),
            )

            self.history_list.addItem(
                item,
            )

    def _load_history_item(
        self,
        item: QListWidgetItem,
    ) -> None:
        session_id = item.data(
            Qt.UserRole,
        )

        if not session_id:
            return

        try:
            replay = ReplayAPI(
                self.storage_root,
            )

            snapshot = replay.session(
                str(session_id),
            )

            self._display_snapshot(
                snapshot,
            )

            self.status_label.setText(f"Loaded session: {session_id}")

            self._set_running_state(
                False,
            )

        except Exception as exc:
            self.status_label.setText(f"Replay error: {exc}")

    # ==================================================================
    # Helpers
    # ==================================================================

    @staticmethod
    def _safe_text(
        value: Any,
    ) -> str:
        if value is None:
            return "<missing>"

        try:
            return str(value)
        except Exception:
            return "<unprintable>"

    # ==================================================================
    # Lifecycle
    # ==================================================================

    def closeEvent(self, event) -> None:
        self.bridge.close()

        super().closeEvent(
            event,
        )
