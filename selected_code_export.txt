from __future__ import annotations

import json
import os
from pathlib import Path
from threading import Lock
from typing import Any

from .models import ObservationSnapshot


class SessionStore:
    """Append-oriented JSONL persistence for completed observation sessions."""

    def __init__(self, root: str | Path) -> None:
        self.root = Path(root)
        self.root.mkdir(parents=True, exist_ok=True)
        self._lock = Lock()

    def save(self, snapshot: ObservationSnapshot) -> Path:
        path = self.root / f"{snapshot.identity.session_id}.json"
        payload = snapshot_to_dict(snapshot)
        temporary = path.with_suffix(".json.tmp")
        with self._lock:
            temporary.write_text(json.dumps(payload, ensure_ascii=False, indent=2), encoding="utf-8")
            os.replace(temporary, path)
        return path

    def load(self, session_id: str) -> ObservationSnapshot:
        path = self.root / f"{session_id}.json"
        return snapshot_from_dict(json.loads(path.read_text(encoding="utf-8")))

    def list_session_ids(self) -> list[str]:
        return sorted(p.stem for p in self.root.glob("*.json"))

    def replay(self, session_id: str) -> ObservationSnapshot:
        return self.load(session_id)


def snapshot_to_dict(snapshot: ObservationSnapshot) -> dict[str, Any]:
    return {
        "identity": {
            "session_id": snapshot.identity.session_id,
            "family_key": snapshot.identity.family_key,
            "task_key": snapshot.identity.task_key,
            "started_at": snapshot.identity.started_at,
        },
        "started_at": snapshot.started_at,
        "finished_at": snapshot.finished_at,
        "status": snapshot.status,
        "metadata": dict(snapshot.metadata),
        "discrete": [item.__dict__ for item in snapshot.discrete],
        "continuous": [item.__dict__ for item in snapshot.continuous],
        "events": [
            {**item.__dict__, "payload": dict(item.payload)}
            for item in snapshot.events
        ],
        "traces": [
            {
                **item.__dict__,
                "discrete": dict(item.discrete),
                "continuous": dict(item.continuous),
            }
            for item in snapshot.traces
        ],
    }


def snapshot_from_dict(data: dict[str, Any]) -> ObservationSnapshot:
    from .models import (
        ContinuousObservation,
        DiscreteObservation,
        TaskEvent,
        TaskIdentity,
        TracePoint,
    )

    identity = TaskIdentity(**data["identity"])
    return ObservationSnapshot(
        identity=identity,
        discrete=tuple(DiscreteObservation(**x) for x in data.get("discrete", [])),
        continuous=tuple(ContinuousObservation(**x) for x in data.get("continuous", [])),
        events=tuple(TaskEvent(**x) for x in data.get("events", [])),
        traces=tuple(TracePoint(**x) for x in data.get("traces", [])),
        started_at=data["started_at"],
        finished_at=data.get("finished_at"),
        status=data.get("status", "unknown"),
        metadata=data.get("metadata", {}),
    )

from __future__ import annotations

from pathlib import Path
from typing import Iterable

from .models import ObservationSnapshot
from .storage import SessionStore


class ReplayAPI:
    """Read-only API for visualization, experiments and mathematical adapters."""

    def __init__(self, root: str | Path) -> None:
        self.store = SessionStore(root)

    def session_ids(self) -> list[str]:
        return self.store.list_session_ids()

    def session(self, session_id: str) -> ObservationSnapshot:
        return self.store.load(session_id)

    def sessions(self, session_ids: Iterable[str] | None = None) -> list[ObservationSnapshot]:
        ids = list(session_ids) if session_ids is not None else self.session_ids()
        return [self.session(session_id) for session_id in ids]

    def feature_rows(self, session_ids: Iterable[str] | None = None) -> list[dict[str, object]]:
        rows: list[dict[str, object]] = []
        for snapshot in self.sessions(session_ids):
            row: dict[str, object] = {
                "session_id": snapshot.identity.session_id,
                "family_key": snapshot.identity.family_key,
                "status": snapshot.status,
            }
            for item in snapshot.discrete:
                row[f"discrete.{item.name}"] = item.value
            for item in snapshot.continuous:
                if item.value is not None:
                    row[f"continuous.{item.name}"] = item.value
            rows.append(row)
        return rows

from __future__ import annotations

from threading import RLock
from typing import Any, Mapping

from .buckets import BucketManager
from .collectors import ContinuousFeatureCollector, DiscreteFeatureCollector, EventTraceCollector
from .models import ObservationSnapshot, TaskIdentity, utc_now_iso
from .storage import SessionStore


class TaskObservationSession:
    """Live observation session owned by a separate observation thread/process."""

    def __init__(self, identity: TaskIdentity, store: SessionStore | None = None, buckets: BucketManager | None = None) -> None:
        self.identity = identity
        self.discrete = DiscreteFeatureCollector()
        self.continuous = ContinuousFeatureCollector()
        self.events = EventTraceCollector()
        self.store = store
        self.buckets = buckets
        self._started_at = identity.started_at
        self._finished_at: str | None = None
        self._status = "running"
        self._metadata: dict[str, Any] = {}
        self._lock = RLock()

    def set_metadata(self, values: Mapping[str, Any]) -> None:
        with self._lock:
            self._metadata.update(values)

    def finish(self, status: str = "completed") -> ObservationSnapshot:
        with self._lock:
            self._finished_at = utc_now_iso()
            self._status = status
            snapshot = self.snapshot()

        if self.buckets is not None:
            self.buckets.add(snapshot)
        if self.store is not None:
            self.store.save(snapshot)
        return snapshot

    def snapshot(self) -> ObservationSnapshot:
        with self._lock:
            return ObservationSnapshot(
                identity=self.identity,
                discrete=self.discrete.snapshot(),
                continuous=self.continuous.snapshot(),
                events=self.events.events(),
                traces=self.events.traces(),
                started_at=self._started_at,
                finished_at=self._finished_at,
                status=self._status,
                metadata=self._metadata,
            )


class TaskObservationEngine:
    """Public facade used by future filesystem/browser adapters."""

    def __init__(self, storage_root: str | None = None) -> None:
        self.store = SessionStore(storage_root) if storage_root else None
        self.buckets = BucketManager()
        self._sessions: dict[str, TaskObservationSession] = {}
        self._lock = RLock()

    def start(self, family_key: str, *, task_key: str = "", metadata: Mapping[str, Any] | None = None) -> TaskObservationSession:
        identity = TaskIdentity.create(family_key, task_key)
        session = TaskObservationSession(identity, self.store, self.buckets)
        if metadata:
            session.set_metadata(metadata)
        with self._lock:
            self._sessions[identity.session_id] = session
        return session

    def get(self, session_id: str) -> TaskObservationSession | None:
        with self._lock:
            return self._sessions.get(session_id)

    def finish(self, session_id: str, status: str = "completed") -> ObservationSnapshot | None:
        session = self.get(session_id)
        if session is None:
            return None
        snapshot = session.finish(status)
        with self._lock:
            self._sessions.pop(session_id, None)
        return snapshot

    def replay(self, session_id: str) -> ObservationSnapshot:
        if self.store is None:
            raise RuntimeError("Persistent storage is not configured.")
        return self.store.replay(session_id)

from __future__ import annotations

from dataclasses import dataclass, field
from datetime import datetime, timezone
from enum import Enum
from types import MappingProxyType
from typing import Any, Mapping
import math
import uuid


def utc_now_iso() -> str:
    return datetime.now(timezone.utc).isoformat()


class ObservationKind(str, Enum):
    SESSION_STARTED = "session_started"
    SESSION_FINISHED = "session_finished"
    FEATURE = "feature"
    EVENT = "event"
    TRACE = "trace"
    DECISION = "decision"
    ERROR = "error"


@dataclass(frozen=True)
class TaskIdentity:
    session_id: str
    family_key: str
    task_key: str = ""
    started_at: str = field(default_factory=utc_now_iso)

    @classmethod
    def create(cls, family_key: str, task_key: str = "") -> "TaskIdentity":
        return cls(
            session_id=uuid.uuid4().hex,
            family_key=family_key,
            task_key=task_key,
        )


@dataclass(frozen=True)
class DiscreteObservation:
    name: str
    value: str | int | bool | None
    timestamp: str = field(default_factory=utc_now_iso)
    source: str = "unknown"
    step: int | None = None


@dataclass(frozen=True)
class ContinuousObservation:
    name: str
    value: float | None
    timestamp: str = field(default_factory=utc_now_iso)
    source: str = "unknown"
    step: int | None = None

    def __post_init__(self) -> None:
        if self.value is not None and not math.isfinite(float(self.value)):
            object.__setattr__(self, "value", None)


@dataclass(frozen=True)
class TaskEvent:
    name: str
    timestamp: str = field(default_factory=utc_now_iso)
    source: str = "unknown"
    payload: Mapping[str, Any] = field(default_factory=dict)
    step: int | None = None

    def __post_init__(self) -> None:
        object.__setattr__(self, "payload", MappingProxyType(dict(self.payload)))


@dataclass(frozen=True)
class TracePoint:
    sequence: int
    timestamp: str
    phase: str
    action: str = ""
    path: str | None = None
    discrete: Mapping[str, str | int | bool | None] = field(default_factory=dict)
    continuous: Mapping[str, float | None] = field(default_factory=dict)

    def __post_init__(self) -> None:
        object.__setattr__(self, "discrete", MappingProxyType(dict(self.discrete)))
        object.__setattr__(self, "continuous", MappingProxyType(dict(self.continuous)))


@dataclass(frozen=True)
class ObservationSnapshot:
    identity: TaskIdentity
    discrete: tuple[DiscreteObservation, ...]
    continuous: tuple[ContinuousObservation, ...]
    events: tuple[TaskEvent, ...]
    traces: tuple[TracePoint, ...]
    started_at: str
    finished_at: str | None
    status: str
    metadata: Mapping[str, Any] = field(default_factory=dict)

    def __post_init__(self) -> None:
        object.__setattr__(self, "metadata", MappingProxyType(dict(self.metadata)))

    def to_feature_vector(self, names: tuple[str, ...]) -> tuple[float, ...]:
        latest: dict[str, float] = {}
        for item in self.continuous:
            if item.value is not None:
                latest[item.name] = float(item.value)
        return tuple(latest.get(name, 0.0) for name in names)


@dataclass(frozen=True)
class BucketKey:
    name: str
    value: str


@dataclass
class ObservationBucket:
    key: BucketKey
    count: int = 0
    sessions: list[str] = field(default_factory=list)

    def add(self, session_id: str) -> None:
        self.count += 1
        if session_id not in self.sessions:
            self.sessions.append(session_id)

from pathlib import Path

from task_observation.database import SQLiteObservationStore


def test_database_survives_store_restart(tmp_path: Path) -> None:
    database_path = tmp_path / "observations.sqlite3"

    with SQLiteObservationStore(database_path) as store:
        store.repository.create_session(
            session_id="session-001",
            task_family="filesystem.read_table.calculate",
            started_at="2026-10-03T10:00:00",
            status="running",
        )

        store.repository.insert_observation(
            session_id="session-001",
            sequence=1,
            timestamp="2026-10-03T10:00:01",
            name="row_count",
            kind="continuous",
            value=1000,
        )

        store.repository.insert_observation(
            session_id="session-001",
            sequence=2,
            timestamp="2026-10-03T10:00:02",
            name="optional_value",
            kind="continuous",
            value=None,
        )

        store.repository.insert_observation(
            session_id="session-001",
            sequence=3,
            timestamp="2026-10-03T10:00:03",
            name="extension",
            kind="discrete",
            value=".csv",
        )

        store.repository.insert_event(
            session_id="session-001",
            sequence=4,
            timestamp="2026-10-03T10:00:04",
            event_type="calculate.completed",
            payload={
                "result": 1044000000,
            },
        )

        store.repository.insert_label(
            session_id="session-001",
            label="ALLOW",
            source="human",
            timestamp="2026-10-03T10:00:05",
        )

        store.repository.finish_session(
            session_id="session-001",
            finished_at="2026-10-03T10:00:06",
            status="completed",
        )

        store.commit()

    assert database_path.exists()

    # Simulate completely new application runtime.
    with SQLiteObservationStore(database_path) as store:
        session = store.repository.get_session("session-001")

        assert session is not None
        assert session.status == "completed"

        observations = store.repository.observations("session-001")

        assert len(observations) == 3
        assert observations[0].value == 1000.0
        assert observations[1].value is None
        assert observations[2].value == ".csv"

        events = store.repository.events("session-001")

        assert len(events) == 1
        assert events[0].payload["result"] == 1044000000

        labels = store.repository.labels("session-001")

        assert len(labels) == 1
        assert labels[0].label == "ALLOW"


def test_database_supports_multiple_sessions(tmp_path: Path) -> None:
    database_path = tmp_path / "observations.sqlite3"

    with SQLiteObservationStore(database_path) as store:
        for index in range(5):
            session_id = f"session-{index}"

            store.repository.create_session(
                session_id=session_id,
                task_family="demo.family",
                started_at=f"2026-10-03T10:00:0{index}",
                status="completed",
            )

        store.commit()

    with SQLiteObservationStore(database_path) as store:
        session_ids = store.repository.list_session_ids()

        assert len(session_ids) == 5
        assert "session-0" in session_ids
        assert "session-4" in session_ids
