from __future__ import annotations

from dataclasses import dataclass, field
from typing import Any, Callable

ToolHandler = Callable[[dict[str, Any]], "ToolResult"]


@dataclass(frozen=True)
class ToolCall:
    """
    Generic representation of one tool invocation.

    This model intentionally knows nothing about Gemini,
    filesystem operations, compute operations, or UI code.
    """

    name: str
    arguments: dict[str, Any] = field(default_factory=dict)


@dataclass
class ToolResult:
    """
    Generic result returned by a tool.

    The dispatcher uses this as the common boundary between
    individual tool implementations and the caller.
    """

    name: str
    success: bool
    data: Any = None
    message: str = ""
    error: str | None = None

    @classmethod
    def ok(
        cls,
        name: str,
        data: Any = None,
        message: str = "",
    ) -> "ToolResult":
        return cls(
            name=name,
            success=True,
            data=data,
            message=message,
            error=None,
        )

    @classmethod
    def failure(
        cls,
        name: str,
        error: str,
    ) -> "ToolResult":
        return cls(
            name=name,
            success=False,
            data=None,
            message="",
            error=error,
        )

    def to_dict(self) -> dict[str, Any]:
        """
        Return a stable dictionary representation suitable for
        serialization and passing back to an external caller.
        """

        return {
            "tool": self.name,
            "success": self.success,
            "data": self.data,
            "message": self.message,
            "error": self.error,
        }


@dataclass(frozen=True)
class ToolSpec:
    """
    Description of one registered tool.

    `declaration` is intentionally represented as a plain dictionary
    so the registry remains independent of any particular AI SDK.
    """

    name: str
    description: str
    declaration: dict[str, Any]
    handler: ToolHandler

    def as_declaration(self) -> dict[str, Any]:
        """
        Return the external tool declaration.

        The declaration can later be converted into the format
        expected by Gemini, another model provider, or another
        tool-calling system.
        """

        return self.declaration

from __future__ import annotations

from collections.abc import Iterable

from .tool_models import ToolHandler, ToolSpec


class ToolRegistryError(RuntimeError):
    """Raised when a tool registry operation is invalid."""


class ToolRegistry:
    """
    Central registry of available tools.

    The registry knows which tools exist, but it does not know
    how Gemini or another model calls them.
    """

    def __init__(self) -> None:
        self._tools: dict[str, ToolSpec] = {}

    def register(
        self,
        *,
        name: str,
        description: str,
        declaration: dict,
        handler: ToolHandler,
    ) -> ToolSpec:
        """
        Register one tool.
        """

        if not name or not name.strip():
            raise ToolRegistryError("Tool name must be a non-empty string.")

        if name in self._tools:
            raise ToolRegistryError(f"Tool is already registered: {name}")

        spec = ToolSpec(
            name=name,
            description=description,
            declaration=declaration,
            handler=handler,
        )

        self._tools[name] = spec
        return spec

    def register_spec(self, spec: ToolSpec) -> ToolSpec:
        """
        Register an already-created ToolSpec.
        """

        if spec.name in self._tools:
            raise ToolRegistryError(f"Tool is already registered: {spec.name}")

        self._tools[spec.name] = spec
        return spec

    def register_many(self, specs: Iterable[ToolSpec]) -> None:
        """
        Register multiple tools.
        """

        for spec in specs:
            self.register_spec(spec)

    def get(self, name: str) -> ToolSpec:
        """
        Return a registered tool.
        """

        try:
            return self._tools[name]
        except KeyError as exc:
            raise ToolRegistryError(f"Unknown tool: {name}") from exc

    def contains(self, name: str) -> bool:
        """
        Return whether a tool is registered.
        """

        return name in self._tools

    def names(self) -> list[str]:
        """
        Return registered tool names in registration order.
        """

        return list(self._tools.keys())

    def specs(self) -> list[ToolSpec]:
        """
        Return all registered tool specifications.
        """

        return list(self._tools.values())

    def declarations(self) -> list[dict]:
        """
        Return declarations for all registered tools.

        The declarations remain plain dictionaries so this class
        remains independent of Gemini's SDK.
        """

        return [spec.as_declaration() for spec in self._tools.values()]

    def __len__(self) -> int:
        return len(self._tools)

    def __contains__(self, name: str) -> bool:
        return self.contains(name)

from __future__ import annotations

from typing import Any

from .tool_models import ToolCall, ToolResult
from .tool_registry import ToolRegistry, ToolRegistryError


class ToolDispatcherError(RuntimeError):
    """Raised when a tool cannot be dispatched."""


class ToolDispatcher:
    """
    Central execution boundary for all registered tools.

    The dispatcher does not implement filesystem or compute logic.

    Its only responsibility is:

        ToolCall
            ↓
        ToolRegistry
            ↓
        registered handler
            ↓
        ToolResult
    """

    def __init__(self, registry: ToolRegistry) -> None:
        self.registry = registry

    def dispatch(self, call: ToolCall) -> ToolResult:
        """
        Execute one generic tool call.
        """

        if not isinstance(call, ToolCall):
            raise TypeError("dispatch() requires a ToolCall.")

        try:
            spec = self.registry.get(call.name)
        except ToolRegistryError as exc:
            return ToolResult.failure(
                name=call.name,
                error=str(exc),
            )

        try:
            arguments = dict(call.arguments or {})

            result = spec.handler(arguments)

            if not isinstance(result, ToolResult):
                return ToolResult.failure(
                    name=call.name,
                    error=(
                        f"Tool handler returned an invalid result type: "
                        f"{type(result).__name__}"
                    ),
                )

            return result

        except Exception as exc:
            return ToolResult.failure(
                name=call.name,
                error=f"{type(exc).__name__}: {exc}",
            )

    def dispatch_name(
        self,
        name: str,
        arguments: dict[str, Any] | None = None,
    ) -> ToolResult:
        """
        Convenience wrapper for callers that already have a tool name
        and argument dictionary.
        """

        return self.dispatch(
            ToolCall(
                name=name,
                arguments=arguments or {},
            )
        )

    def declarations(self) -> list[dict]:
        """
        Return all registered tool declarations.

        The caller can adapt these declarations to Gemini or another
        model provider.
        """

        return self.registry.declarations()

    def available_tools(self) -> list[str]:
        """
        Return the names of all currently available tools.
        """

        return self.registry.names()

    @staticmethod
    def _operation(
        operation: OperationType,
        arguments: dict[str, Any],
    ) -> FilesystemOperation:
        """
        Convert generic tool arguments into the existing filesystem
        domain model.

        Validation and sandbox enforcement remain downstream in the
        filesystem service.
        """

        return FilesystemOperation(
            operation=operation,
            path=arguments.get("path"),
            destination=arguments.get("destination"),
            pattern=arguments.get("pattern"),
            query=arguments.get("query"),
            content=arguments.get("content"),
            expression=arguments.get("expression"),
            recursive=bool(arguments.get("recursive", False)),
            encoding=arguments.get("encoding", "utf-8"),
            options=arguments.get("options", {}),
        )

    def _execute(
        self,
        operation: FilesystemOperation,
    ) -> ToolResult:
        """
        Execute through SandboxFilesystemService.

        No filesystem Path operation is performed here.
        """

        result = self.service.execute(operation)

        if not result.success:
            return ToolResult.failure(
                name=operation.operation.value,
                error=result.message,
            )

        return ToolResult.ok(
            name=operation.operation.value,
            data=self._serialize(result.data),
            message=result.message,
        )

    # ==================================================================
    # Serialization
    # ==================================================================

    @classmethod
    def _serialize(cls, value: Any) -> Any:
        if value is None:
            return None

        if is_dataclass(value):
            return cls._serialize(asdict(value))

        if isinstance(value, Enum):
            return value.value

        if isinstance(value, list):
            return [cls._serialize(item) for item in value]

        if isinstance(value, tuple):
            return [cls._serialize(item) for item in value]

        if isinstance(value, dict):
            return {str(key): cls._serialize(item) for key, item in value.items()}

        if isinstance(value, (str, int, float, bool)):
            return value

        return str(value)

from __future__ import annotations

from typing import Any

from ..compute.compute_models import ComputeOperation, ComputeRequest

class ComputeToolAdapter:
    """
    Adapter between generic tool calls and ComputeService.

    This adapter contains no Gemini or UI logic.
    """

                self._declaration_min(),
                self.minimum,

    def sum(self, arguments: dict[str, Any]) -> ToolResult:
        return self._execute(
            tool_name="compute_sum",
            operation=ComputeOperation.SUM,
            arguments=arguments,
        )

    def average(self, arguments: dict[str, Any]) -> ToolResult:
        return self._execute(
            tool_name="compute_average",
            operation=ComputeOperation.AVERAGE,
            arguments=arguments,
        )

    def minimum(self, arguments: dict[str, Any]) -> ToolResult:
        return self._execute(
            tool_name="compute_min",
            operation=ComputeOperation.MIN,
            arguments=arguments,
        )

    def maximum(self, arguments: dict[str, Any]) -> ToolResult:
        return self._execute(
            tool_name="compute_max",
            operation=ComputeOperation.MAX,
            arguments=arguments,
        )

    def count(self, arguments: dict[str, Any]) -> ToolResult:
        return self._execute(
            tool_name="compute_count",
            operation=ComputeOperation.COUNT,
            arguments=arguments,
        )

    # ==================================================================
    # Compute boundary
    # ==================================================================

    def _execute(
        self,
        *,
        tool_name: str,
        operation: ComputeOperation,
        arguments: dict[str, Any],
    ) -> ToolResult:
        values = arguments.get("values")

        if not isinstance(values, list):
            return ToolResult.failure(
                name=tool_name,
                error="Tool argument 'values' must be a list.",
            )

        request = ComputeRequest(
            operation=operation,
            values=values,
        )

        result = self.service.execute(request)

        if not result.success:
            return ToolResult.failure(
                name=tool_name,
                error=result.error or "Compute operation failed.",
            )

        return ToolResult.ok(
            name=tool_name,
            data=result.value,
        )
