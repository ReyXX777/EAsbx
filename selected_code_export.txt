from __future__ import annotations

from typing import Any

from PySide6.QtCore import QObject, QThread, Signal, Slot

from .observation_thread import ObservationWorker


class ObservationBridge(QObject):
    """Thread-safe bridge between Qt UI code and ObservationWorker.

    UI code talks to this class.

    UI code never directly accesses the TaskObservationSession.
    """

    task_started = Signal(str)
    observation_updated = Signal(object)
    task_finished = Signal(object)
    error = Signal(str)

    request_start = Signal(str)
    request_discrete = Signal(str, object)
    request_continuous = Signal(str, object)
    request_event = Signal(str, object)
    request_snapshot = Signal()
    request_finish = Signal()
    request_shutdown = Signal()

    def __init__(
        self,
        storage_root: str,
        parent: QObject | None = None,
    ):
        super().__init__(parent)

        self._thread = QThread(self)

        self._worker = ObservationWorker(
            storage_root,
        )

        self._worker.moveToThread(
            self._thread,
        )

        self.request_start.connect(
            self._worker.start_task,
        )

        self.request_discrete.connect(
            self._worker.record_discrete,
        )

        self.request_continuous.connect(
            self._worker.record_continuous,
        )

        self.request_event.connect(
            self._worker.record_event,
        )

        self.request_snapshot.connect(
            self._worker.publish_current_state,
        )

        self.request_finish.connect(
            self._worker.finish_task,
        )

        self.request_shutdown.connect(
            self._worker.shutdown,
        )

        self._worker.task_started.connect(
            self.task_started,
        )

        self._worker.observation_updated.connect(
            self.observation_updated,
        )

        self._worker.task_finished.connect(
            self.task_finished,
        )

        self._worker.worker_error.connect(
            self.error,
        )

        self._thread.start()

    def start_task(self, family_key: str) -> None:
        self.request_start.emit(
            family_key,
        )

    def record_discrete(
        self,
        name: str,
        value: Any = None,
    ) -> None:
        self.request_discrete.emit(
            name,
            value,
        )

    def record_continuous(
        self,
        name: str,
        value: Any = None,
    ) -> None:
        self.request_continuous.emit(
            name,
            value,
        )

    def record_event(
        self,
        event_type: str,
        payload: Any = None,
    ) -> None:
        self.request_event.emit(
            event_type,
            payload,
        )

    def request_current_snapshot(self) -> None:
        self.request_snapshot.emit()

    def finish_task(self) -> None:
        self.request_finish.emit()

    def close(self) -> None:
        """Stop the observation worker safely."""

        if not self._thread.isRunning():
            return

        self.request_shutdown.emit()

        self._thread.quit()

        if not self._thread.wait(3000):
            self._thread.terminate()
            self._thread.wait(1000)

    def __del__(self):
        try:
            self.close()
        except Exception:
            pass

from __future__ import annotations

from typing import Any

from PySide6.QtCore import QObject, Signal, Slot

from ..session import TaskObservationEngine


class ObservationWorker(QObject):
    """Owns mutable observation state on a dedicated worker thread.

    No Qt widget should directly manipulate a TaskObservationSession.

    The worker receives requests through queued Qt signals and emits
    immutable-ish snapshot objects back to the presentation layer.
    """

    task_started = Signal(str)
    observation_updated = Signal(object)
    task_finished = Signal(object)
    worker_error = Signal(str)

    def __init__(self, storage_root: str, parent: QObject | None = None):
        super().__init__(parent)

        self._engine = TaskObservationEngine(storage_root)
        self._session = None

    @Slot(str)
    def start_task(self, family_key: str) -> None:
        """Start a new observation session."""

        try:
            if self._session is not None:
                self.worker_error.emit(
                    "Cannot start a new task while another task is active."
                )
                return

            family_key = (family_key or "").strip()

            if not family_key:
                self.worker_error.emit(
                    "Cannot start an observation without a task family."
                )
                return

            self._session = self._engine.start(family_key)

            self.task_started.emit(family_key)

        except Exception as exc:
            self.worker_error.emit(f"{type(exc).__name__}: {exc}")

    @Slot(str, object)
    def record_discrete(self, name: str, value: Any) -> None:
        """Record one discrete feature.

        Missing values are deliberately allowed.
        """

        if self._session is None:
            self.worker_error.emit("No active observation session.")
            return

        try:
            self._session.discrete.record(
                name,
                value,
            )

        except Exception as exc:
            self.worker_error.emit(
                f"Discrete observation error: {type(exc).__name__}: {exc}"
            )

    @Slot(str, object)
    def record_continuous(self, name: str, value: Any) -> None:
        """Record one continuous feature.

        Invalid/missing values are allowed to reach the collector, which
        already defines the prototype's non-halting behaviour.
        """

        if self._session is None:
            self.worker_error.emit("No active observation session.")
            return

        try:
            self._session.continuous.record(
                name,
                value,
            )

        except Exception as exc:
            self.worker_error.emit(
                f"Continuous observation error: " f"{type(exc).__name__}: {exc}"
            )

    @Slot(str, object)
    def record_event(self, event_type: str, payload: Any = None) -> None:
        """Record a generic observation event when the session supports it.

        Stage 6B intentionally does not assume a particular event collector
        implementation. This keeps the Qt layer compatible with the current
        collector implementation while allowing richer event models later.
        """

        if self._session is None:
            self.worker_error.emit("No active observation session.")
            return

        try:
            collector = getattr(self._session, "events", None)

            if collector is None:
                collector = getattr(self._session, "trace", None)

            if collector is None:
                # Event collection is optional at this stage.
                return

            record_method = getattr(collector, "record", None)

            if record_method is None:
                return

            record_method(
                event_type,
                payload,
            )

        except Exception as exc:
            self.worker_error.emit(
                f"Event observation error: " f"{type(exc).__name__}: {exc}"
            )

    @Slot()
    def publish_current_state(self) -> None:
        """Publish the current state when the session implementation supports it.

        The existing session API guarantees finish(), but not necessarily
        an intermediate snapshot API. Therefore this method is deliberately
        conservative.
        """

        if self._session is None:
            return

        try:
            snapshot_method = getattr(
                self._session,
                "snapshot",
                None,
            )

            if snapshot_method is None:
                return

            snapshot = snapshot_method()

            self.observation_updated.emit(snapshot)

        except Exception as exc:
            self.worker_error.emit(f"Snapshot error: {type(exc).__name__}: {exc}")

    @Slot()
    def finish_task(self) -> None:
        """Finish the active observation session."""

        if self._session is None:
            self.worker_error.emit("No active observation session.")
            return

        try:
            snapshot = self._session.finish()

            self._session = None

            self.task_finished.emit(snapshot)
            self.observation_updated.emit(snapshot)

        except Exception as exc:
            self.worker_error.emit(
                f"Task finalization error: " f"{type(exc).__name__}: {exc}"
            )

    @Slot()
    def shutdown(self) -> None:
        """Release the active session reference."""

        self._session = None

from __future__ import annotations

import math

from PySide6.QtCore import QPointF, Qt
from PySide6.QtGui import QPainter, QPen
from PySide6.QtWidgets import QWidget


class ObservationPlot(QWidget):
    """Small dependency-free Qt plot for continuous observation traces."""

    def __init__(self, parent=None):
        super().__init__(parent)

        self.setMinimumHeight(220)

        self._values: list[float] = []
        self._title = "Continuous Feature"

    def set_data(
        self,
        values: list[float],
        title: str = "Continuous Feature",
    ) -> None:
        self._values = [
            float(value)
            for value in values
            if value is not None and math.isfinite(float(value))
        ]

        self._title = title

        self.update()

    def clear(self) -> None:
        self._values.clear()
        self.update()

    def paintEvent(self, _event) -> None:
        painter = QPainter(self)

        try:
            painter.setRenderHint(
                QPainter.Antialiasing,
                True,
            )

            rect = self.rect().adjusted(
                45,
                25,
                -20,
                -35,
            )

            painter.drawText(
                10,
                18,
                self._title,
            )

            if not self._values:
                painter.drawText(
                    rect,
                    Qt.AlignCenter,
                    "No numeric observations.",
                )
                return

            minimum = min(self._values)
            maximum = max(self._values)

            if math.isclose(minimum, maximum):
                minimum -= 1.0
                maximum += 1.0

            painter.drawRect(rect)

            point_count = len(self._values)

            points: list[QPointF] = []

            for index, value in enumerate(self._values):

                if point_count == 1:
                    x = rect.center().x()
                else:
                    x = rect.left() + (index / (point_count - 1)) * rect.width()

                normalized = (value - minimum) / (maximum - minimum)

                y = rect.bottom() - normalized * rect.height()

                points.append(
                    QPointF(
                        x,
                        y,
                    )
                )

            if len(points) >= 2:
                painter.drawPolyline(
                    points,
                )
            else:
                painter.drawEllipse(
                    points[0],
                    3,
                    3,
                )

            painter.drawText(
                5,
                rect.top(),
                f"{maximum:.4g}",
            )

            painter.drawText(
                5,
                rect.bottom(),
                f"{minimum:.4g}",
            )

            painter.drawText(
                rect.left(),
                rect.bottom() + 25,
                "start",
            )

            painter.drawText(
                rect.right() - 35,
                rect.bottom() + 25,
                "end",
            )

        finally:
            painter.end()

    def _render_selected_continuous(
        self,
        observations,
    ) -> None:
        selected = self.feature_selector.currentText()

        if not selected:
            self.plot.clear()
            return

        values: list[float] = []

        for observation in observations:

            name = str(
                getattr(
                    observation,
                    "name",
                    "",
                )
            )

            if name != selected:
                continue

            value = getattr(
                observation,
                "value",
                None,
            )

            try:
                if value is not None:
                    values.append(float(value))
            except (
                TypeError,
                ValueError,
            ):
                continue

        self.plot.set_data(
            values,
            selected,
        )

                self.storage_root,
            )

            session_ids = replay.session_ids()

        except Exception as exc:
            self.status_label.setText(f"History error: {exc}")
            return

        self.history_list.clear()

        for session_id in session_ids:

            item = QListWidgetItem(
                str(session_id),
            )

            item.setData(
                Qt.UserRole,
                str(session_id),
            )

            self.history_list.addItem(
                item,
            )

    def _load_history_item(
        self,
        item: QListWidgetItem,
    ) -> None:
        session_id = item.data(
            Qt.UserRole,
        )

        if not session_id:
            return

        try:
            replay = ReplayAPI(
                self.storage_root,
            )

            snapshot = replay.session(
                session_id,
            )

            self._display_snapshot(
                snapshot,
            )

            self.status_label.setText(f"Loaded session: {session_id}")

        except Exception as exc:
            self.status_label.setText(f"Replay error: {exc}")

    # ---------------------------------------------------------
    # Helpers
    # ---------------------------------------------------------

    @staticmethod
    def _safe_text(
        value: Any,
    ) -> str:
        if value is None:
            return "<missing>"

        try:
            return str(value)
        except Exception:
            return "<unprintable>"

    # ---------------------------------------------------------
    # Lifecycle
    # ---------------------------------------------------------

    def closeEvent(self, event) -> None:
        self.bridge.close()

        super().closeEvent(
            event,
        )
