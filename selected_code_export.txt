from __future__ import annotations

from threading import RLock
from typing import Any, Mapping

from .buckets import BucketManager
from .collectors import (
    ContinuousFeatureCollector,
    DiscreteFeatureCollector,
    EventTraceCollector,
)
from .models import ObservationSnapshot, TaskIdentity, utc_now_iso
from .storage import SessionStore


class TaskObservationSession:
    """Live observation session owned by a separate observation thread/process."""

    def __init__(
        self,
        identity: TaskIdentity,
        store: SessionStore | None = None,
        buckets: BucketManager | None = None,
    ) -> None:
        self.identity = identity

        self.discrete = DiscreteFeatureCollector()
        self.continuous = ContinuousFeatureCollector()
        self.events = EventTraceCollector()

        self.store = store
        self.buckets = buckets

        self._started_at = identity.started_at
        self._finished_at: str | None = None
        self._status = "running"
        self._metadata: dict[str, Any] = {}

        self._lock = RLock()

    def set_metadata(self, values: Mapping[str, Any]) -> None:
        with self._lock:
            self._metadata.update(values)

    def finish(
        self,
        status: str = "completed",
    ) -> ObservationSnapshot:
        with self._lock:
            self._finished_at = utc_now_iso()
            self._status = status
            snapshot = self.snapshot()

        if self.buckets is not None:
            self.buckets.add(snapshot)

        if self.store is not None:
            self.store.save(snapshot)

        return snapshot

    def snapshot(self) -> ObservationSnapshot:
        with self._lock:
            return ObservationSnapshot(
                identity=self.identity,
                discrete=self.discrete.snapshot(),
                continuous=self.continuous.snapshot(),
                events=self.events.events(),
                traces=self.events.traces(),
                started_at=self._started_at,
                finished_at=self._finished_at,
                status=self._status,
                metadata=self._metadata,
            )


class TaskObservationEngine:
    """Public facade used by filesystem, browser, and other adapters."""

    def __init__(
        self,
        storage_root: str | None = None,
    ) -> None:
        self.store = SessionStore(storage_root) if storage_root else None

        self.buckets = BucketManager()

        self._sessions: dict[
            str,
            TaskObservationSession,
        ] = {}

        self._lock = RLock()

    def start(
        self,
        family_key: str,
        *,
        task_key: str = "",
        metadata: Mapping[str, Any] | None = None,
    ) -> TaskObservationSession:
        identity = TaskIdentity.create(
            family_key,
            task_key,
        )

        session = TaskObservationSession(
            identity,
            self.store,
            self.buckets,
        )

        if metadata:
            session.set_metadata(metadata)

        with self._lock:
            self._sessions[identity.session_id] = session

        return session

    def get(
        self,
        session_id: str,
    ) -> TaskObservationSession | None:
        with self._lock:
            return self._sessions.get(session_id)

    def finish(
        self,
        session_id: str,
        status: str = "completed",
    ) -> ObservationSnapshot | None:
        session = self.get(session_id)

        if session is None:
            return None

        snapshot = session.finish(status)

        with self._lock:
            self._sessions.pop(
                session_id,
                None,
            )

        return snapshot

    def replay(
        self,
        session_id: str,
    ) -> ObservationSnapshot:
        if self.store is None:
            raise RuntimeError("Persistent storage is not configured.")

        return self.store.replay(session_id)

from __future__ import annotations

from threading import Lock
from typing import Any, Mapping
from datetime import datetime, timezone

from .models import ContinuousObservation, DiscreteObservation, TaskEvent, TracePoint


class DiscreteFeatureCollector:
    """Thread-safe collector for categorical/state-like task features."""

    def __init__(self) -> None:
        self._items: list[DiscreteObservation] = []
        self._lock = Lock()

    def record(self, name: str, value: str | int | bool | None, *, source: str = "unknown", step: int | None = None) -> None:
        with self._lock:
            self._items.append(DiscreteObservation(name, value, source=source, step=step))

    def record_many(self, values: Mapping[str, str | int | bool | None], *, source: str = "unknown", step: int | None = None) -> None:
        with self._lock:
            for name, value in values.items():
                self._items.append(DiscreteObservation(name, value, source=source, step=step))

    def snapshot(self) -> tuple[DiscreteObservation, ...]:
        with self._lock:
            return tuple(self._items)

    def clear(self) -> None:
        with self._lock:
            self._items.clear()


class ContinuousFeatureCollector:
    """Thread-safe collector for numeric task features.

    Missing/invalid numeric values are accepted and represented as None rather
    than terminating the observation process.
    """

    def __init__(self) -> None:
        self._items: list[ContinuousObservation] = []
        self._lock = Lock()

    def record(self, name: str, value: float | int | None, *, source: str = "unknown", step: int | None = None) -> None:
        safe_value: float | None
        try:
            safe_value = None if value is None else float(value)
        except (TypeError, ValueError):
            safe_value = None
        with self._lock:
            self._items.append(ContinuousObservation(name, safe_value, source=source, step=step))

    def record_many(self, values: Mapping[str, float | int | None], *, source: str = "unknown", step: int | None = None) -> None:
        for name, value in values.items():
            self.record(name, value, source=source, step=step)

    def snapshot(self) -> tuple[ContinuousObservation, ...]:
        with self._lock:
            return tuple(self._items)

    def clear(self) -> None:
        with self._lock:
            self._items.clear()


class EventTraceCollector:
    """Thread-safe event and trace recorder."""

    def __init__(self) -> None:
        self._events: list[TaskEvent] = []
        self._traces: list[TracePoint] = []
        self._lock = Lock()

    def event(self, name: str, *, source: str = "unknown", payload: Mapping[str, Any] | None = None, step: int | None = None) -> None:
        with self._lock:
            self._events.append(TaskEvent(name=name, source=source, payload=payload or {}, step=step))

    def trace(
        self,
        phase: str,
        *,
        action: str = "",
        path: str | None = None,
        discrete: Mapping[str, str | int | bool | None] | None = None,
        continuous: Mapping[str, float | None] | None = None,
    ) -> int:
        with self._lock:
            sequence = len(self._traces)
            self._traces.append(
                TracePoint(
                    sequence=sequence,
                    timestamp=datetime.now(timezone.utc).isoformat(),
                    phase=phase,
                    action=action,
                    path=path,
                    discrete=discrete or {},
                    continuous=continuous or {},
                )
            )
            return sequence

    def events(self) -> tuple[TaskEvent, ...]:
        with self._lock:
            return tuple(self._events)

    def traces(self) -> tuple[TracePoint, ...]:
        with self._lock:
            return tuple(self._traces)

from __future__ import annotations

from collections import Counter
from dataclasses import dataclass
from typing import Any

from ..models import ObservationSnapshot


@dataclass(frozen=True)
class ToolObservation:
    """Derived observations about tool usage within a task."""

    total_tool_calls: int
    successful_tool_calls: int
    failed_tool_calls: int
    unique_tools: int

    list_directory_count: int
    search_files_count: int
    read_file_count: int
    read_table_count: int
    compute_count: int

    tool_counts: dict[str, int]


class ToolObserver:
    """Derives tool-usage observations from an ObservationSnapshot."""

    def observe(self, snapshot: ObservationSnapshot) -> ToolObservation:
        tool_started = [
            event for event in snapshot.events if event.name == "tool_started"
        ]

        tool_completed = [
            event for event in snapshot.events if event.name == "tool_completed"
        ]

        tool_counts: Counter[str] = Counter()

        for event in tool_started:
            tool_name = event.payload.get("tool_name")

            if tool_name:
                tool_counts[str(tool_name)] += 1

        successful = sum(
            1 for event in tool_completed if event.payload.get("success") is True
        )

        failed = sum(
            1 for event in tool_completed if event.payload.get("success") is False
        )

        return ToolObservation(
            total_tool_calls=len(tool_started),
            successful_tool_calls=successful,
            failed_tool_calls=failed,
            unique_tools=len(tool_counts),
            list_directory_count=tool_counts.get("list_directory", 0),
            search_files_count=tool_counts.get("search_files", 0),
            read_file_count=tool_counts.get("read_file", 0),
            read_table_count=tool_counts.get("read_table", 0),
            compute_count=sum(
                count
                for name, count in tool_counts.items()
                if name.startswith("compute")
            ),
            tool_counts=dict(tool_counts),
        )

from __future__ import annotations

from dataclasses import dataclass
from pathlib import PurePosixPath

from ..models import ObservationSnapshot


@dataclass(frozen=True)
class NavigationObservation:
    """Derived observations about filesystem navigation."""

    unique_paths: int
    paths_visited: int
    directory_navigation_count: int
    file_access_count: int
    maximum_path_depth: int

    search_count: int
    navigation_reversal_count: int

    visited_paths: tuple[str, ...]


class NavigationObserver:
    """Derives navigation-related observations."""

    def observe(self, snapshot: ObservationSnapshot) -> NavigationObservation:
        traces = [trace for trace in snapshot.traces if trace.path is not None]

        paths = [trace.path for trace in traces if trace.path is not None]

        unique_paths = tuple(dict.fromkeys(paths))

        action_phases = {
            "tool_started",
        }

        directory_navigation_count = sum(
            1
            for trace in traces
            if (trace.phase in action_phases and trace.action == "list_directory")
        )

        file_access_count = sum(
            1
            for trace in traces
            if (
                trace.phase in action_phases
                and trace.action
                in {
                    "read_file",
                    "read_table",
                }
            )
        )

        search_count = sum(
            1
            for trace in traces
            if (trace.phase in action_phases and trace.action == "search_files")
        )
        maximum_path_depth = 0

        for path in unique_paths:
            depth = self._path_depth(path)
            maximum_path_depth = max(maximum_path_depth, depth)

        navigation_reversal_count = self._count_reversals(traces)

        return NavigationObservation(
            unique_paths=len(unique_paths),
            paths_visited=len(paths),
            directory_navigation_count=directory_navigation_count,
            file_access_count=file_access_count,
            maximum_path_depth=maximum_path_depth,
            search_count=search_count,
            navigation_reversal_count=navigation_reversal_count,
            visited_paths=unique_paths,
        )

    @staticmethod
    def _path_depth(path: str) -> int:
        normalized = path.replace("\\", "/").strip("/")

        if not normalized or normalized == ".":
            return 0

        return len(
            [part for part in PurePosixPath(normalized).parts if part not in {".", ""}]
        )

    @staticmethod
    def _count_reversals(traces) -> int:
        paths = [trace.path for trace in traces if trace.path is not None]

        reversals = 0

        for previous, current in zip(paths, paths[1:]):
            if previous == current:
                continue

            previous_parts = previous.replace("\\", "/").strip("/").split("/")
            current_parts = current.replace("\\", "/").strip("/").split("/")

            if (
                len(current_parts) < len(previous_parts)
                and previous_parts[: len(current_parts)] == current_parts
            ):
                reversals += 1

        return reversals

from __future__ import annotations

from dataclasses import dataclass
from datetime import datetime

from ..models import ObservationSnapshot


@dataclass(frozen=True)
class TimingObservation:
    """Derived timing information."""

    task_duration_seconds: float | None
    average_tool_duration_seconds: float | None
    minimum_tool_duration_seconds: float | None
    maximum_tool_duration_seconds: float | None


class TimingObserver:
    """Derives task and tool timing from event timestamps."""

    def observe(self, snapshot: ObservationSnapshot) -> TimingObservation:
        task_duration = self._duration(
            snapshot.started_at,
            snapshot.finished_at,
        )

        started: dict[tuple[str, str | None], datetime] = {}
        durations: list[float] = []

        for event in snapshot.events:
            tool_name = event.payload.get("tool_name")
            path = event.payload.get("path")

            key = (str(tool_name), path)

            if event.name == "tool_started" and tool_name:
                started[key] = self._parse(event.timestamp)

            elif event.name == "tool_completed" and tool_name:
                start = started.pop(key, None)

                if start is not None:
                    end = self._parse(event.timestamp)
                    duration = (end - start).total_seconds()

                    if duration >= 0:
                        durations.append(duration)

        return TimingObservation(
            task_duration_seconds=task_duration,
            average_tool_duration_seconds=(
                sum(durations) / len(durations) if durations else None
            ),
            minimum_tool_duration_seconds=(min(durations) if durations else None),
            maximum_tool_duration_seconds=(max(durations) if durations else None),
        )

    @staticmethod
    def _parse(value: str) -> datetime:
        return datetime.fromisoformat(value.replace("Z", "+00:00"))

    @classmethod
    def _duration(
        cls,
        started_at: str | None,
        finished_at: str | None,
    ) -> float | None:
        if not started_at or not finished_at:
            return None

        duration = (cls._parse(finished_at) - cls._parse(started_at)).total_seconds()

        return duration if duration >= 0 else None

from __future__ import annotations

from dataclasses import dataclass

from ..models import ObservationSnapshot


@dataclass(frozen=True)
class OutcomeObservation:
    """Derived observations about task outcome."""

    status: str
    completed: bool
    failed: bool
    cancelled: bool

    approval_label: bool | None

    had_tool_failure: bool
    recovered_from_failure: bool


class OutcomeObserver:
    """Derives outcome and recovery observations."""

    def observe(self, snapshot: ObservationSnapshot) -> OutcomeObservation:
        completed = snapshot.status == "completed"
        failed = snapshot.status == "failed"
        cancelled = snapshot.status == "cancelled"

        tool_completed = [
            event for event in snapshot.events if event.name == "tool_completed"
        ]

        had_tool_failure = any(
            event.payload.get("success") is False for event in tool_completed
        )

        recovered_from_failure = self._detect_recovery(snapshot)

        approval_label = self._approval_label(snapshot)

        return OutcomeObservation(
            status=snapshot.status,
            completed=completed,
            failed=failed,
            cancelled=cancelled,
            approval_label=approval_label,
            had_tool_failure=had_tool_failure,
            recovered_from_failure=recovered_from_failure,
        )

    @staticmethod
    def _approval_label(
        snapshot: ObservationSnapshot,
    ) -> bool | None:
        for item in reversed(snapshot.discrete):
            if item.name in {
                "approval_label",
                "approved",
                "human_approved",
            }:
                if item.value is None:
                    return None

                return bool(item.value)

        return None

    @staticmethod
    def _detect_recovery(
        snapshot: ObservationSnapshot,
    ) -> bool:
        failure_seen = False

        for event in snapshot.events:
            if event.name != "tool_completed":
                continue

            if event.payload.get("success") is False:
                failure_seen = True
                continue

            if failure_seen and event.payload.get("success") is True:
                return True

        return False

        if self.timing.task_duration_seconds is not None:

            "recovered_from_failure": (self.outcome.recovered_from_failure),
            "used_search": self.tool.search_files_count > 0,
            "used_computation": self.tool.compute_count > 0,
            "used_file_read": (
                self.tool.read_file_count > 0 or self.tool.read_table_count > 0
            ),
        }

        if self.outcome.approval_label is not None:
            values["approval_label"] = self.outcome.approval_label

        return values


class FeatureDeriver:
    """Coordinates all independent observation derivations."""

    def __init__(
        self,
        *,
        tool_observer: ToolObserver | None = None,
        navigation_observer: NavigationObserver | None = None,
        timing_observer: TimingObserver | None = None,
        outcome_observer: OutcomeObserver | None = None,
    ) -> None:
        self.tool_observer = tool_observer or ToolObserver()
        self.navigation_observer = navigation_observer or NavigationObserver()
        self.timing_observer = timing_observer or TimingObserver()
        self.outcome_observer = outcome_observer or OutcomeObserver()

    def derive(
        self,
        snapshot: ObservationSnapshot,
    ) -> DerivedObservationFeatures:
        return DerivedObservationFeatures(
            tool=self.tool_observer.observe(snapshot),
            navigation=self.navigation_observer.observe(snapshot),
            timing=self.timing_observer.observe(snapshot),
            outcome=self.outcome_observer.observe(snapshot),
        )

            name="tool_completed",
            timestamp="2026-10-03T13:03:42+00:00",
            source="filesystem_worker",
            payload={
                "tool_name": "read_file",
                "path": "recovered.txt",
                "success": False,
            },
        ),
    )

    events.insert(
        -1,
        TaskEvent(
            name="tool_completed",
            timestamp="2026-10-03T13:03:42.500000+00:00",
            source="filesystem_worker",
            payload={
                "tool_name": "read_file",
                "path": "recovered.txt",
                "success": True,
            },
        ),
    )

    modified = ObservationSnapshot(
        identity=snapshot.identity,
        discrete=snapshot.discrete,
        continuous=snapshot.continuous,
        events=tuple(events),
        traces=snapshot.traces,
        started_at=snapshot.started_at,
        finished_at=snapshot.finished_at,
        status=snapshot.status,
        metadata=snapshot.metadata,
    )

    observation = OutcomeObserver().observe(modified)

    assert observation.had_tool_failure is True
    assert observation.recovered_from_failure is True

from pathlib import Path

from task_observation import ReplayAPI, TaskObservationEngine, generate_runs


def test_missing_values_do_not_stop_collection(tmp_path: Path) -> None:
    engine = TaskObservationEngine(tmp_path)
    session = engine.start("demo.family")
    session.discrete.record("optional", None)
    session.continuous.record("optional_number", None)
    session.continuous.record("bad_number", "not-a-number")
    snapshot = session.finish()

    assert snapshot.status == "completed"
    assert len(snapshot.discrete) == 1
    assert len(snapshot.continuous) == 2
    assert snapshot.continuous[-1].value is None


def test_persistence_and_replay(tmp_path: Path) -> None:
    engine = TaskObservationEngine(tmp_path)
    session = engine.start("filesystem.read_table.calculate")
    session.continuous.record("row_count", 1000)
    session.discrete.record("extension", ".csv")
    snapshot = session.finish()

    replay = ReplayAPI(tmp_path)
    loaded = replay.session(snapshot.identity.session_id)
    assert loaded.identity.session_id == snapshot.identity.session_id
    assert loaded.continuous[0].value == 1000.0
    assert loaded.discrete[0].value == ".csv"


def test_simulation_produces_multiple_replayable_runs(tmp_path: Path) -> None:
    snapshots = generate_runs(tmp_path, count=5, seed=42)
    assert len(snapshots) == 5
    replay = ReplayAPI(tmp_path)
    assert len(replay.session_ids()) == 5
    assert len(replay.feature_rows()) == 5
